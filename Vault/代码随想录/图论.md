# 理论基础
#### 图的种类
有向图：图中边是有方向的
无向图：图中边无方向
加权有\无向图：图中边是有权值的
#### 度
无向图中有几个边连接该节点
有向图： 入度：指向该节点的边的个数
		出度：从该节点出发的边的个数
#### 连通性
无向图中任两个节点都是可达的，称为连通图
有节点达不到其他节点，称为非连通图
**强连通图**：有向图中，任两点都是可达的
**连通分量**：无向图中，**极大**连通子图
**强连通分量**：有向图中，极大强连通子图
# 图的存储
## 邻接矩阵
n个节点申请n\*n的二维数组，
`g[u][v] = 0`表示节点u到v无边（不能直接表示权值为 0 的边）
`g[u][v] = a`表示节点u到v有边，且权值是a
## 邻接表
n个节点申请n个容器（比如双向链表）的一维数组，
`g[u]` 存一个容器，容器里存从u直接可达的邻居信息
# 题目
## 力扣题目
### 所有可能的路径
vector容器的邻接表实现图
```cpp
class Solution {
public:
    vector<int> path;
    vector<vector<int>> res;
    void dfs(const vector<vector<int>>& graph, int x, int n){
        if(x == n){
            res.push_back(path);
            return;
        }
        for(int i = 0; i < graph[x].size(); i++){
            path.push_back(graph[x][i]);
            dfs(graph, graph[x][i], n);
            path.pop_back();
        }
    }
    vector<vector<int>> allPathsSourceTarget(vector<vector<int>>& graph) {
        path.push_back(0);
        dfs(graph, 0, graph.size()-1);
        return res;
    }
};
```
## 卡码网题目
### 98可达路径
邻接矩阵版本（本题图无环，DFS 不需要额外 visited）
```cpp
#include<iostream>
#include<vector>
using namespace std;
vector<vector<int>> res;
vector<int> path;
//graph：当前图，x：当前节点，n：终点
void dfs(const vector<vector<int>>& graph, int x, int n){
    if(x == n){
        res.push_back(path);
        return;
    }
    for(int i = 0; i <= n; i++){
        if(graph[x][i] == 1){
            path.push_back(i);
            dfs(graph, i, n);
            path.pop_back();
        }
    }        
}
int main(){
    int n, m, s, t;
    cin >> n >> m;
    //邻接矩阵
    vector<vector<int>> graph(n+1, vector<int>(n+1, 0));
    while(m--){
        cin >> s >> t;
        graph[s][t] = 1;
    }
    path.push_back(1);
    dfs(graph, 1, n);
    if(res.size() == 0)cout<< -1 <<endl;
    for(const vector<int> &pa : res){
        for(int i = 0; i < pa.size()-1; i++){
            cout << pa[i] << " ";
        }
        cout << pa[pa.size()-1] << endl;
    }
}
```
邻接表版本
```cpp
#include<iostream>
#include<vector>
#include<list>
using namespace std;
vector<vector<int>> res;
vector<int> path;
void dfs(const vector<list<int>>& graph, int x, int n){
    if(x == n){
        res.push_back(path);
        return;
    }
    for(int i : graph[x]){
        path.push_back(i);
        dfs(graph, i, n);
        path.pop_back();
    }
}
int main(){
    int n, m, s, t;
    cin >> n >> m;
    vector<list<int>> graph(n+1);
    while(m--){
        cin >> s >> t;
        graph[s].push_back(t);
    }
    path.push_back(1);
    dfs(graph, 1, n);
    if(res.empty())cout << -1 << endl;
    for(int i = 0; i < res.size(); i++){
        for(int j = 0; j < res[i].size()-1; j++){
            cout << res[i][j] << " ";
        }
        cout << res[i][res[i].size()-1] << endl;
    }
}
```
