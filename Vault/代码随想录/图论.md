# 理论基础
#### 图的种类
有向图：图中边是有方向的
无向图：图中边无方向
加权有\无向图：图中边是有权值的
#### 度
无向图中有几个边连接该节点
有向图： 入度：指向该节点的边的个数
		出度：从该节点出发的边的个数
#### 连通性
无向图中任两个节点都是可达的，称为连通图
有节点达不到其他节点，称为非连通图
**强连通图**：有向图中，任两点都是可达的
**连通分量**：无向图中，**极大**连通子图
**强连通分量**：有向图中，极大强连通子图
# 图的存储
## 邻接矩阵
n个节点申请n\*n的二维数组，
`g[u][v] = 0`表示节点u到v无边（不能直接表示权值为 0 的边）
`g[u][v] = a`表示节点u到v有边，且权值是a
## 邻接表
n个节点申请n个容器（比如双向链表）的一维数组，
`g[u]` 存一个容器，容器里存从u直接可达的邻居信息

```cpp
#include<iostream>
#include<vector>
#include<queue>
using namespace std;
/*
   dir[0] = {0, 1} dx = 0, dy = 1 行不变，列+1 向右
   dir[1] = {1, 0} 行+1，列不变 向下
   dir[2] = {-1, 0}向上
   dir[3] = {0, -1}向左
*/
int dir[4][2] = {0, 1, 1, 0, -1, 0, 0, -1};
void bfs(const vector<vector<int>>& grid, vector<vector<bool>>& visited, int x, int y){
    queue<pair<int, int>> que;
    que.push({x, y});
    visited[x][y] = true;
    while(!que.empty()){
        pair<int, int> cur = que.front();
        que.pop();
        int curx = cur.first;
        int cury = cur.second;
        for(int i = 0; i < 4; i++){
            int nextx = curx + dir[i][0];
            int nexty = cury + dir[i][1];
            if(nextx < 0 || nextx >= grid.size() || nexty < 0 || nexty >= grid[0].size())continue;
            if(visited[nextx][nexty] == false && grid[nextx][nexty] == 1){
                que.push({nextx, nexty});
                visited[nextx][nexty] = true;
            }
        }
    }
}
int main(){
    int n, m;
    cin >> n >> m;
    vector<vector<int>> grid(n, vector<int>(m));
    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            cin >> grid[i][j];
        }
    }

    int res = 0;
    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            if(visited[i][j] == false && grid[i][j] == 1){
                res++;
                bfs(grid, visited, i, j);
            }
        }
    }
    cout << res;
}
```

```cpp
#include<iostream>
#include<vector>
using namespace std;
int dir[4][2] = {0, 1, 1, 0, -1, 0, 0, -1};
void dfs(const vector<vector<int>>& grid, vector<vector<bool>>& can, int x, int y){
    can[x][y] = true;
    for(int i = 0; i <4; i++){
        int nextx = x + dir[i][0];
        int nexty = y + dir[i][1];
        if(nextx < 0 ||nextx >= grid.size() || nexty < 0 || nexty >= grid[0].size())continue;
        if(!can[nextx][nexty] && grid[x][y] <= grid[nextx][nexty]){
            dfs(grid, can, nextx, nexty);
        }
    }
}
```

### 104建造最大岛屿
```cpp
#include<iostream>
#include<vector>
#include<algorithm>
#include<unordered_map>
#include<unordered_set>
using namespace std;
int dir[4][2] = {0, 1, 1, 0, -1, 0, 0, -1};
unordered_map<int, int> mp;//key为岛屿编号，值为岛屿大小
void dfs(vector<vector<int>>& grid, int x, int y,int key, int& val){
    grid[x][y] = key;
    val++;
    for(int i = 0; i < 4; i++){
        int nextx = x + dir[i][0];
        int nexty = y + dir[i][1];
        if(nextx < 0 || nextx >= grid.size() || nexty < 0 || nexty >= grid[0].size())continue;
        if(grid[nextx][nexty] == 1)
            dfs(grid, nextx, nexty, key, val);
    }

}
int main(){
    int n, m;
    cin >> n >> m;
    vector<vector<int>> grid(n, vector<int>(m));
    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            cin >> grid[i][j];
        }
    }
    int key = 2;
    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            if(grid[i][j] == 1){
                int val = 0;
                dfs(grid, i, j, key, val);
                mp[key++] = val;
            }
        }
    }
    int res = 0;
    //处理整张地图没有水的情况
    for(auto it : mp)res = max(res, it.second);
    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            if(grid[i][j] == 0){
                //用set去重岛屿编号
                unordered_set<int> st;
                int temp = 1;
                for(int k = 0; k < 4; k++){
                    int nextx = i + dir[k][0];
                    int nexty = j + dir[k][1];
                    if(nextx < 0 || nextx >= grid.size() || nexty < 0 || nexty >= grid[0].size())continue;
                    if(grid[nextx][nexty] != 0){
                        st.insert(grid[nextx][nexty]);
                    }
                }
                for(auto it : st)temp += mp[it];
                if(temp > res)res = temp;
            }
        }
    }
    cout << res;
}
```

### 最大人工岛
```cpp
class Solution {
public:
    int dir[4][2] = {0, 1, 1, 0, -1, 0, 0, -1};
    void dfs(vector<vector<int>>& grid, int x, int y, int key, int& val){
        grid[x][y] = key;
        val++;
        for(int i = 0; i < 4; i++){
            int nx = x + dir[i][0];
            int ny = y + dir[i][1];
            if(nx < 0 || nx >= grid.size() || ny < 0 || ny >= grid[0].size())continue;
            if(grid[nx][ny] == 1){
                dfs(grid, nx, ny, key, val);
            }
        }
    }
    int largestIsland(vector<vector<int>>& grid) {
        unordered_map<int, int> mp;
        int key = 2;
        for(int i = 0; i < grid.size(); i++){
            for(int j = 0; j < grid[0].size(); j++){
                if(grid[i][j] == 1){
                    int val = 0;
                    dfs(grid, i, j, key, val);
                    mp[key++] = val;
                }
            }
        }
        int res = 0;
        for(auto it : mp){
            if(res < it.second) res = it.second;
        }
        for(int i = 0; i < grid.size(); i++){
            for(int j = 0; j < grid[0].size(); j++){
                if(grid[i][j] != 0)continue;
                unordered_set<int> st;
                int temp = 1;
                for(int k = 0; k < 4; k++){
                    int nx = i + dir[k][0];
                    int ny = j + dir[k][1];
                    if(nx < 0 || nx >= grid.size() || ny < 0 || ny >= grid[0].size())continue;
                    if(grid[nx][ny] != 0){
                        st.insert(grid[nx][ny]);
                    }
                }
                for(auto it : st)temp += mp[it];
                if(res < temp)res = temp;
            }
        }
        return res;
    }
};
```
