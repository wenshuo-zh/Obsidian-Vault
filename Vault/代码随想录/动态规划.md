dp四步解题法：
1. 定义dp数组含义
2. 确定递推公式（推导状态转移方程）
3. dp数组初始化
4. 确定遍历顺序

# 基础题目
### 斐波那契数
```cpp
class Solution {
public:
    int fib(int n) {
        if(n <= 1)return n;
        //1.dp数组含义：下标i的斐波那契数值
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = dp[i - 1] + dp[i - 2];
        //3.初始化
        dp[0] = 0;
        dp[1] = 1;
        //4.遍历顺序：从前向后
        for(int i = 2; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```
### 爬楼梯
```cpp
class Solution {
public:
    int climbStairs(int n) {
        if(n <= 2) return n;
        //1.dp数组含义：到达第i阶的方法数
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = dp[i-1] + dp[i-2]
        //因为到i阶可以从i-1阶走一阶或者从i-2阶走两阶（从i-2走两个一阶被从i-1阶的包含过）
        //3.初始化
        dp[1] = 1;
        dp[2] = 2;
        //4.遍历顺序：从前向后
        for(int i = 3; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```
### 使用最小花费爬楼梯
```cpp
class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
    //1.dp数组含义：到达第i阶台阶顶部的最小花费
    vector<int> dp(cost.size() + 1);
    //2.递推公式：dp[i] = min(dp[i - 1] + cost[i - 1], dp[i - 2] + cost[i - 2])
    //3.初始化
    dp[0] = 0;
    dp[1] = 0;
    //4.顺序：从前向后
    for (int i = 2; i <= cost.size(); i++) {
        dp[i] = min(cost[i - 1] + dp[i - 1], cost[i - 2] + dp[i - 2]);
    }
    return dp[cost.size()];
    }
};
```
### 不同路径
```cpp
class Solution {
public:
    int uniquePaths(int m, int n) {
        //1.dp数组的含义：从起点移动到下标i，j的路径数
        vector<vector<int>> dp(m+1, vector<int>(n+1));
        //2.递推公式：dp[i][j] = dp[i - 1][j] + dp[i][j - 1]
	        //（正上方向下走一步 + 正左侧向右走一步）
        //3.初始化
        for(int i = 1; i <= m; i++)dp[i][1] = 1;
        for(int j = 1; j <= n; j++)dp[1][j] = 1;
        //4.遍历方向：从前向后
        for(int i = 2; i <= m; i++){
            for(int j = 2; j <= n; j++){
                dp[i][j] = dp[i-1][j] + dp[i][j-1];
            }
        }
        return dp[m][n];
    }
};
```
### 不同路径II
踩坑：初始化边界时遇到障碍，只是把当前值设成0，后面的也都走不到了都要为0
```cpp
class Solution {
public:
    int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
        int m = obstacleGrid.size();
        int n = obstacleGrid[0].size();
        //1.dp数组含义：到达(i,j)位置的路径数
        vector<vector<int>> dp(m, vector<int>(n));
        //2. 递推公式：dp[i][j] = dp[i-1][j] + dp[i][j -1]
        //3.初始化
        dp[0][0] = obstacleGrid[0][0] == 0 ? 1 : 0;
        for(int i = 1; i < m; i++){
            if(obstacleGrid[i][0] == 1)dp[i][0] = 0;
            else dp[i][0] = dp[i-1][0];
        }
        for(int j = 1; j < n; j++){
            if(obstacleGrid[0][j] == 1)dp[0][j] = 0;
            else dp[0][j] = dp[0][j-1];
        }
        //4.递推方向：从前向后
        for(int i = 1; i < m; i++){
         for(int j = 1; j < n; j++){
             if (obstacleGrid[i][j] == 1) {
                 dp[i][j] = 0;
             } else {
                 dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
             }
         }
        }
        return dp[m-1][n-1];
    }
};
```
### 整数拆分
`dp[i]` 表示把整数 `i` 拆成至少两个正整数后的最大乘积。先拆成 `j` 和 `i-j`：`j * (i-j)` 表示剩余的 `i-j` 不再拆，`j * dp[i-j]` 表示继续拆剩余部分；两种情况取最大值。
```cpp
class Solution {
public:
    int integerBreak(int n) {
        //1.dp数组含义：整数i拆成至少两个正整数后的最大乘积
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = max(j*(i - j), j*dp[i - j])
        //3.初始化
        dp[0] = 0;
        dp[1] = 0; // 1 无法拆成两个正整数，本题不会使用 dp[1]
        dp[2] = 1;
        //4.从前向后遍历
        for(int i = 3;i <= n; i++){
            dp[i] = 0;
            for(int j = 1; j <= i / 2; j++){
                int temp = max(j * (i - j), j * dp[i - j]);
                if(temp > dp[i])dp[i] = temp;
            }
        }
        return dp[n];
    }
};
```
### 不同的二叉搜索树
j是根节点的值，j-1是左子树的节点个数，i-j是右子树的节点个数
```cpp
class Solution {
public:
    int numTrees(int n) {
        //1.dp含义：i个节点组成的二叉搜索树有几种
        vector<int> dp(n+1);
        //2.递推公式：j从1到i dp[i] += dp[j-1] * dp[i-j]
            //j是根节点的值，j-1是左子树的节点个数，i-j是右子树的节点个数
        //3.初始化
        dp[0] = 1;
        //4.从前向后遍历
        for(int i = 1; i <= n; i++){
            for(int j = 1; j <= i; j++){
                dp[i] += dp[j-1] * dp[i-j];
            }
        }
        return dp[n];
    }
};
```

==1.dp数组的含义==：
物品i，背包重量j
**dp\[i]\[j]** 表示从下标为\[0-i]的物品里任意取，放进容量为j的背包，价值总和最大是多少。

| i \ j | 0   | 1   | 2   | 3   | 4   |
| ----- | --- | --- | --- | --- | --- |
| 物品0   | 0   | 15  | 15  | 15  | 15  |
| 物品1   | 0   | 15  | 15  | 20  | 35  |
| 物品2   | 0   | 15  | 15  | 20  | 35  |
==2.确定递推公式==：
`dp[i][j] = max(dp[i-1][j], dp[i-1][j-weight(i)]+value[i]);
dp\[i-1]\[j-weight(i)]表示从\[0 ~ i-1]个物品里任取，放已经放i的剩余容量的背包里的最大价值
==3.初始化==：

| i \ j | 0   | 1   | 2   | 3   | 4   |
| ----- | --- | --- | --- | --- | --- |
| 物品0   | 0   | 15  | 15  | 15  | 15  |
| 物品1   | 0   |     |     |     |     |
| 物品2   | 0   |     |     |     |     |
==4.遍历顺序==
```cpp
for (int i = 1; i < weight.size(); i++) {
    for (int j = 0; j <= target; j++) {
        if (j < weight[i]) {
            dp[i][j] = dp[i - 1][j];
        } else {
            dp[i][j] = max(dp[i - 1][j],
```

### 卡码网46 携带研究材料
#### 二维dp数组版本：
```cpp
#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
    int m, n;
    cin >> m >> n;
    vector<int> weight(m, 0), value(m);
    for(int i = 0; i < m; i++)cin >> weight[i];
    for(int i = 0; i < m; i++)cin >> value[i];
    //1.定义dp数组：dp[i][j]从下标为[0-i]的物品里任意取，放进容量为j的背包，价值总和最大是多少。
    vector<vector<int>> dp(m, vector<int>(n+1));
    //2.递推公式：dp[i][j] = max(dp[i-1][j], dp[i-1][j-weight(i)]+value[i])
    //3.初始化
    for(int j = 0; j <= n; j++){
        if(weight[0] <= j)dp[0][j] = value[0];
        else dp[0][j] = 0;
    }
    //4.遍历顺序
    for(int i = 1; i < m; i++){
        for(int j = 0; j <= n; j++){
            if(j-weight[i] < 0)dp[i][j] = dp[i-1][j];
            else dp[i][j] = max(dp[i-1][j], dp[i-1][j-weight[i]] + value[i]);
        }
    }
    cout<< dp[m-1][n] << endl;
    return 0;
}
```
#### 一维dp滚动数组版本
注意踩坑点：j一定要倒序遍历，防止读取 `j - weight[i]` 时拿到本层的数据；倒序遍历保证读取到的是上一轮的数据。
```cpp
#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
    int m, n;
    cin >> m >> n;
    vector<int> weight(m, 0), value(m);
    for(int i = 0; i < m; i++)cin >> weight[i];
    for(int i = 0; i < m; i++)cin >> value[i];
    //1.定义dp数组：dp[j]容量为j的背包，价值总和最大是多少。
    vector<int> dp(n+1);
    //2.递推公式：dp[i][j] = max(dp[i-1][j], dp[i-1][j-weight(i)]+value[i])
    //3.初始化
    //4.遍历顺序：容量j倒序遍历。计算大j时，j-weight[i]还没有在本轮被修改，保留的是上一轮的数据，等价二维dp[i-1][j-weight[i]]
    for(int i = 0; i < m; i++){
        for(int j = n; j >= weight[i]; j--){
            dp[j] = max(dp[j], dp[j - weight[i]] + value[i]);
        }
    }
    cout<< dp[n] << endl;
    return 0;
}
```
### 分割等和子集
找集合里能否出现和为sum/2的子集，数字的重量和价值都是数字的本身值，看看能不能装满容量为sum/2的背包，且一个数字只能用一次，转换为01背包问题
```cpp
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int target = 0;
        for(int i = 0; i < nums.size(); i++)target += nums[i];
        if(target % 2 == 1)return false;
        target /= 2;
        //1.dp数组含义：从前面已经遍历过的数字里选若干个，所选数字总和≤j，能凑出来的最大总和
        vector<int> dp(target+1, 0);
        //2.递推公式：dp[j] = max(dp[j], dp[j-nums[i]] + nums[i]);
        //3.初始化
        //4.遍历顺序：先正序遍历数字再倒序遍历容量
        for(int i = 0; i < nums.size(); i++){
            for(int j = target; j >= nums[i]; j--){
                dp[j] = max(dp[j], dp[j-nums[i]] + nums[i]);
            }
        }
        if(dp[target] == target)return true;
        else return false;
    }
};
```
