dp四步解题法：
1. 定义dp数组含义
2. 确定递推公式（推导状态转移方程）
3. dp数组初始化
4. 确定遍历顺序

# 基础题目
## 斐波那契数
```cpp
class Solution {
public:
    int fib(int n) {
        if(n <= 1)return n;
        //1.dp数组含义：下标i的斐波那契数值
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = dp[i - 1] + dp[i - 2];
        //3.初始化
        dp[0] = 0;
        dp[1] = 1;
        //4.遍历顺序：从前向后
        for(int i = 2; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```
## 爬楼梯
```cpp
class Solution {
public:
    int climbStairs(int n) {
        if(n <= 2) return n;
        //1.dp数组含义：到达第i阶的方法数
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = dp[i-1] + dp[i-2]
        //因为到i阶可以从i-1阶走一阶或者从i-2阶走两阶（从i-2走两个一阶被从i-1阶的包含过）
        //3.初始化
        dp[1] = 1;
        dp[2] = 2;
        //4.遍历顺序：从前向后
        for(int i = 3; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```
## 使用最小花费爬楼梯
```cpp
class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
    //1.dp数组含义：到达第i阶台阶顶部的最小花费
    vector<int> dp(cost.size() + 1);
    //2.递推公式：dp[i] = min(dp[i - 1] + cost[i - 1], dp[i - 2] + cost[i - 2])
    //3.初始化
    dp[0] = 0;
    dp[1] = 0;
    //4.顺序：从前向后
    for (int i = 2; i <= cost.size(); i++) {
        dp[i] = min(cost[i - 1] + dp[i - 1], cost[i - 2] + dp[i - 2]);
    }
    return dp[cost.size()];
    }
};
```
