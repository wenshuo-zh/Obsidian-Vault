dp四步解题法：
1. 定义dp数组含义
2. 确定递推公式（推导状态转移方程）
3. dp数组初始化
4. 确定遍历顺序

# 基础题目
### 斐波那契数
```cpp
class Solution {
public:
    int fib(int n) {
        if(n <= 1)return n;
        //1.dp数组含义：下标i的斐波那契数值
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = dp[i - 1] + dp[i - 2];
        //3.初始化
        dp[0] = 0;
        dp[1] = 1;
        //4.遍历顺序：从前向后
        for(int i = 2; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```

### 使用最小花费爬楼梯
```cpp
class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
    //1.dp数组含义：到达第i阶台阶顶部的最小花费
    vector<int> dp(cost.size() + 1);
    //2.递推公式：dp[i] = min(dp[i - 1] + cost[i - 1], dp[i - 2] + cost[i - 2])
    //3.初始化
    dp[0] = 0;
    dp[1] = 0;
    //4.顺序：从前向后
    for (int i = 2; i <= cost.size(); i++) {
        dp[i] = min(cost[i - 1] + dp[i - 1], cost[i - 2] + dp[i - 2]);
    }
    return dp[cost.size()];
    }
};
```

### 不同路径

```cpp
class Solution {
public:
    int uniquePaths(int m, int n) {
        //1.dp数组的含义：从起点移动到下标i，j的路径数
        vector<vector<int>> dp(m+1, vector<int>(n+1));
        //2.递推公式：dp[i][j] = dp[i - 1][j] + dp[i][j - 1]
	        //（正上方向下走一步 + 正左侧向右走一步）
        //3.初始化
        for(int i = 1; i <= m; i++)dp[i][1] = 1;
        for(int j = 1; j <= n; j++)dp[1][j] = 1;
        //4.遍历方向：从前向后
        for(int i = 2; i <= m; i++){
            for(int j = 2; j <= n; j++){
                dp[i][j] = dp[i-1][j] + dp[i][j-1];
            }
        }
        return dp[m][n];
    }
};
```

### 不同路径II
踩坑：初始化边界时遇到障碍，只是把当前值设成0，后面的也都走不到了都要为0
```cpp
class Solution {
public:
    int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
        int m = obstacleGrid.size();
        int n = obstacleGrid[0].size();
        //1.dp数组含义：到达(i,j)位置的路径数
        vector<vector<int>> dp(m, vector<int>(n));
        //2. 递推公式：dp[i][j] = dp[i-1][j] + dp[i][j -1]
        //3.初始化
        dp[0][0] = obstacleGrid[0][0] == 0 ? 1 : 0;
        for(int i = 1; i < m; i++){
            if(obstacleGrid[i][0] == 1)dp[i][0] = 0;
            else dp[i][0] = dp[i-1][0];
        }
        for(int j = 1; j < n; j++){
            if(obstacleGrid[0][j] == 1)dp[0][j] = 0;
            else dp[0][j] = dp[0][j-1];
        }
        //4.递推方向：从前向后
        for(int i = 1; i < m; i++){
         for(int j = 1; j < n; j++){
             if (obstacleGrid[i][j] == 1) {
                 dp[i][j] = 0;
             } else {
                 dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
             }
         }
        }
        return dp[m-1][n-1];
    }
};
```
### 整数拆分
`dp[i]` 表示把整数 `i` 拆成至少两个正整数后的最大乘积。先拆成 `j` 和 `i-j`：`j * (i-j)` 表示剩余的 `i-j` 不再拆，`j * dp[i-j]` 表示继续拆剩余部分；两种情况取最大值。
```cpp
class Solution {
public:
    int integerBreak(int n) {
        //1.dp数组含义：整数i拆成至少两个正整数后的最大乘积
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = max(j*(i - j), j*dp[i - j])
        //3.初始化
        dp[0] = 0;
        dp[1] = 0; // 1 无法拆成两个正整数，本题不会使用 dp[1]
        dp[2] = 1;
        //4.从前向后遍历
        for(int i = 3;i <= n; i++){
            dp[i] = 0;
            for(int j = 1; j <= i / 2; j++){
                int temp = max(j * (i - j), j * dp[i - j]);
                if(temp > dp[i])dp[i] = temp;
            }
        }
        return dp[n];
    }
};
```
### 不同的二叉搜索树
j是根节点的值，j-1是左子树的节点个数，i-j是右子树的节点个数
```cpp
class Solution {
public:
    int numTrees(int n) {
        //1.dp含义：i个节点组成的二叉搜索树有几种
        vector<int> dp(n+1);
        //2.递推公式：j从1到i dp[i] += dp[j-1] * dp[i-j]
            //j是根节点的值，j-1是左子树的节点个数，i-j是右子树的节点个数
        //3.初始化
        dp[0] = 1;
        //4.从前向后遍历
        for(int i = 1; i <= n; i++){
            for(int j = 1; j <= i; j++){
                dp[i] += dp[j-1] * dp[i-j];
            }
        }
        return dp[n];
    }
};
```

#### 二维dp数组版本：

```cpp
#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
    int m, n;
    cin >> m >> n;
    vector<int> weight(m, 0), value(m);
    for(int i = 0; i < m; i++)cin >> weight[i];
    for(int i = 0; i < m; i++)cin >> value[i];

    //3.初始化
```

#### 一维dp滚动数组版本
注意踩坑点：j一定要倒序遍历，防止读取 `j - weight[i]` 时拿到本层的数据；倒序遍历保证读取到的是上一轮的数据。
```cpp
#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
    int m, n;
    cin >> m >> n;
    vector<int> weight(m, 0), value(m);
    for(int i = 0; i < m; i++)cin >> weight[i];
    for(int i = 0; i < m; i++)cin >> value[i];
    //1.定义dp数组：dp[j]容量为j的背包，价值总和最大是多少。
    vector<int> dp(n+1);
    //2.递推公式：dp[i][j] = max(dp[i-1][j], dp[i-1][j-weight(i)]+value[i])
    //3.初始化
    //4.遍历顺序：容量j倒序遍历。计算大j时，j-weight[i]还没有在本轮被修改，保留的是上一轮的数据，等价二维dp[i-1][j-weight[i]]
    for(int i = 0; i < m; i++){
        for(int j = n; j >= weight[i]; j--){
            dp[j] = max(dp[j], dp[j - weight[i]] + value[i]);
        }
    }
    cout<< dp[n] << endl;
    return 0;
}
```
### 分割等和子集
找集合里能否出现和为sum/2的子集，数字的重量和价值都是数字的本身值，看看能不能装满容量为sum/2的背包，且一个数字只能用一次，转换为01背包问题
```cpp
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int target = 0;
        for(int i = 0; i < nums.size(); i++)target += nums[i];
        if(target % 2 == 1)return false;
        target /= 2;
        //1.dp数组含义：从前面已经遍历过的数字里选若干个，所选数字总和≤j，能凑出来的最大总和
        vector<int> dp(target+1, 0);
        //2.递推公式：dp[j] = max(dp[j], dp[j-nums[i]] + nums[i]);
        //3.初始化
        //4.遍历顺序：先正序遍历数字再倒序遍历容量
        for(int i = 0; i < nums.size(); i++){
            for(int j = target; j >= nums[i]; j--){
                dp[j] = max(dp[j], dp[j-nums[i]] + nums[i]);
            }
        }
        if(dp[target] == target)return true;
        else return false;
    }
};
```

```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int sum = 0;
        for(int i = 0; i < nums.size(); i++)sum += nums[i];
        if(abs(target) > sum) return 0;
        //1.dp[i][j]数组含义：前i个数添加正负号，运算结果得到j的方案数目
        vector<vector<int>> dp(nums.size(), vector<int>(2*sum+1, 0));
        //2.递推公式：dp[i][j] = dp[i - 1][j - nums[i]] + dp[i - 1][j + nums[i]];
        //3.初始化：
        dp[0][sum + nums[0]] += 1;
        dp[0][sum - nums[0]] += 1;
        //4.遍历顺序：从前向后
        for(int i = 1; i < nums.size(); i++){
            for(int j = 0; j < 2*sum+1; j++){
                if(j - nums[i] >= 0)dp[i][j] += dp[i - 1][j - nums[i]];
                if(j + nums[i] <= 2*sum) dp[i][j] += dp[i - 1][j + nums[i]];
            }
        }
        return dp[nums.size() - 1][target + sum];
    }
};
```

```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        //转换成01背包：从nums任选若干，每个数字用一次，恰好装满容量(sum+target)/2的背包的方案数        
        int sum = 0;
        for(int i = 0; i < nums.size(); i++)sum += nums[i];
        if(abs(target) > sum)return 0;
        if((sum + target) % 2 != 0)return 0;
        int left = (sum + target) / 2;
        //1.定义dp数组：使用下标[0,i]的nums[i]能凑满容量j的背包的方案数
        vector<vector<int>> dp(nums.size(), vector<int>(left + 1, 0));
        //2.确定递推公式：dp[i][j] = dp[i-1][j-nums[i]]（放下标i） + dp[i-1][j]（不放下标i）
        //3.初始化：
        dp[0][0] = 1;
        if(nums[0] <= left)dp[0][nums[0]] += 1;
        //4.遍历：
        for(int i = 1; i < nums.size(); i++){
            for(int j = 0; j <= left; j++){
                dp[i][j] = dp[i-1][j];
                if(j >= nums[i])dp[i][j] += dp[i-1][j-nums[i]];
            }
        }
        return dp[nums.size()-1][left];
    }
};
```
### 一和零
字符串数组中每一个字符串为物品，重量是二维的，分别是0和1的个数；每一个字符串价值是1。找m和n容量背包最多能装下多少个字符串：转换成01背包问题
>用滚动数组实现👇，避免开三维数组
```cpp
class Solution {
public:
    int findMaxForm(vector<string>& strs, int m, int n) {
        //1.dp[i][j]滚动数组：最多用i个0，j个1，最多字符串数
        vector<vector<int>> dp(m+1, vector<int>(n+1, 0));
        //2.递推公式：dp[i][j] = max(dp[i][j], dp[i - cnt0][j - cnt1] + 1);
        //3.初始化：
        //4.遍历顺序：先正序遍历物品（字符串），在倒叙遍历容量（0和1），滚动数组
        for(string &s : strs){
            int cnt0 = 0, cnt1 = 0;
            for(int i = 0; i < s.size(); i++){
                if(s[i] == '0')cnt0++;
                else cnt1++;
            }
            for(int i = m; i >=cnt0; i--){
                for(int j = n; j >= cnt1; j--){
                    dp[i][j] = max(dp[i][j], dp[i - cnt0][j - cnt1] + 1);
                }
            }
        }
        return dp[m][n];
    }
};
```

### 打家劫舍II
房子排成环形，拆分成两段直线形式的打家劫舍，一段不偷首，一段不偷尾，两段最大即答案
```cpp
class Solution {
public:
    //在[begin, end]一排房子里，返回打家劫舍的最大金额
    int robRange(vector<int>& nums, int begin, int end){
        if(begin == end)return nums[begin];
        vector<int> dp(end+1);
        dp[begin] = nums[begin];
        dp[begin+1] = max(nums[begin], nums[begin+1]);
        for(int i = begin+2; i <= end; i++){
            dp[i] = max(dp[i-2] + nums[i], dp[i-1]);
        }
        return dp[end];
    }
    int rob(vector<int>& nums) {
        if(nums.size() == 0)return 0;
        if(nums.size() == 1)return nums[0];
        int res1 = robRange(nums, 0, nums.size()-2);//不偷最后一间
        int res2 = robRange(nums, 1, nums.size()-1);//不偷第一间
        return max(res1, res2);
    }
};
```

# 股票问题
### 买卖股票的最佳时机
`dp[i][0] = max(dp[i-1][0], -prices[i])`：今天持有股票，要么延续昨天持有的状态，要么今天首次买入。
`dp[i][1] = max(dp[i-1][1], dp[i-1][0] + prices[i])`：今天无股票，要么继续空仓，要么今天卖出昨天持有的股票。
```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        //1.dp[i][0]：第i天持有股票所得最多现金
        //  dp[i][1]：第i天不持有股票所得最多现金
        vector<vector<int>> dp(prices.size(), vector<int>(2));
        //2.递推公式：dp[i][0] = max(dp[i-1][0], -prices[i]);
        //           dp[i][1] = max(dp[i-1][1], prices[i] + dp[i-1][0]);
        //3.初始化：
        dp[0][0] = -1 * prices[0];
        dp[0][1] = 0;
        //4.遍历顺序：
        for(int i = 1; i < prices.size(); i++){
            dp[i][0] = max(dp[i-1][0], -prices[i]);
            dp[i][1] = max(dp[i-1][1], prices[i] + dp[i-1][0]);
        }
        return dp[prices.size()-1][1];
    }
};
```

`dp[i][0] = max(dp[i-1][0], dp[i-1][1]-prices[i])`：要么维持昨天持有的状态，要么之前可能有利润+今天新买股票。

`j`代表本轮开始不持有股票的状态，`j+1`代表本轮持有股票，`j+2`是本轮卖出股票后

```cpp
        vector<vector<int>> dp(prices.size(), vector<int>(2*k+1, 0));
        //2.递推公式：dp[i][j+1] = max(dp[i-1][j+1], dp[i-1][j] - prices[i]);

        }

        for(int i = 1; i < prices.size(); i++){
            for(int j = 0; j < 2*k-1; j += 2){
                dp[i][j+1] = max(dp[i-1][j+1], dp[i-1][j] - prices[i]);
                dp[i][j+2] = max(dp[i-1][j+2], dp[i-1][j+1] + prices[i]);
            }
        }
```

```cpp
public:
    int findLengthOfLCIS(vector<int>& nums) {

            if(nums[i] > nums[i-1])dp[i] = dp[i-1] + 1;
            else dp[i] = 1;//断了初始成1
            if(dp[i] > res)res = dp[i];
        }
        return res;
    }
};
```

### 最长公共子序列
双序列，不连续
```cpp
class Solution {
public:
    int longestCommonSubsequence(string text1, string text2) {

        vector<vector<int>> dp(text1.size()+1, vector<int>(text2.size()+1, 0));

        for(int i = 1; i <= text1.size(); i++){
            for(int j = 1; j <= text2.size(); j++){
                if(text1[i-1] == text2[j-1])dp[i][j] = dp[i-1][j-1] + 1;
                else dp[i][j] = max(dp[i-1][j], dp[i][j-1]);//断了继承左边或上方的最优解
                if(res < dp[i][j])res = dp[i][j];
            }

    }
};
```

### 总结一下上面四道题：
区别：连续与否，单数组还是双数组
连续问题：断开后通常重置状态；非连续问题：可以跳过元素，继承之前范围内的最优解
单：继承上一项； 双：继承左上最优

### 不同的子序列
```cpp
class Solution {
public:
    int numDistinct(string s, string t) {
        //dp[i][j]:下标i-1结尾的s中出现下标j-1结尾的t的子序列个数
        vector<vector<unsigned long long>> dp(s.size()+1, vector<unsigned long long>(t.size()+1));
        //初始化：
        for(int i = 0; i <= s.size(); i++)dp[i][0] = 1;
        for(int j = 1; j <= t.size(); j++)dp[0][j] = 0;
        for(int i = 1; i <= s.size(); i++){
            for(int j = 1; j <= t.size(); j++){
                //字符相等：选当前字符和跳过当前字符的结果和
                if(s[i-1] == t[j-1])dp[i][j] = dp[i-1][j-1] + dp[i-1][j];
                //字符不相等：只能跳过当前字符
                else dp[i][j] = dp[i-1][j];
            }
        }
        return dp[s.size()][t.size()];
    }
```

### 两个字符串的删除操作
dp法一：思路和上一题大致，但两个字符串都可以删除，遇到不相等的字符取最小的步数+1即可
```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        //dp[i][j]:下标i-1结尾的word1和下标j-1结尾的word2得到相同的最小步数
        vector<vector<int>> dp(word1.size()+1, vector<int>(word2.size()+1));
        for(int i = 0; i <= word1.size(); i++)dp[i][0] = i;
        for(int j = 0; j <= word2.size(); j++)dp[0][j] = j;
        for(int i = 1; i <= word1.size(); i++){
            for(int j = 1; j <= word2.size(); j++){
                if(word1[i-1] == word2[j-1])dp[i][j] = dp[i-1][j-1];
                else{
                    dp[i][j] = min(dp[i-1][j], dp[i][j-1]) + 1;
                }
            }
        }
        return dp[word1.size()][word2.size()];
    }
};
```
dp法2：求出两个字符串的最长公共子序列，在删除其余字符即可
```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        //找最长的公共子序列（双数组不连续）
        //dp[i][j]下标i-1的word1，下标j-1的word2，最长公共子序列的长度
        vector<vector<int>> dp(word1.size()+1, vector<int>(word2.size()+1));
        int res = 0;
        dp[0][0] = 0;
        for(int i = 1; i <= word1.size(); i++){
            for(int j = 1; j <= word2.size(); j++){
                if(word1[i-1] == word2[j-1])dp[i][j] = dp[i-1][j-1] + 1;
                else dp[i][j] = max(dp[i][j-1], dp[i-1][j]);
                if(res < dp[i][j])res = dp[i][j];
            }
        }
        return word1.size() + word2.size() - 2 * res;
    }
};
```
### 编辑距离
字符不相等时：三种操作选步数最小的
	`dp[i-1][j]`删除操作：word1前i-1字符已成功变成word2前j个字符，删除word1第i个字符
	`dp[i][j-1]`插入操作：word1前i字符已成功变成word2前j-1个字符，插入word2第j个字符
	`dp[i-1][j-1]`替换操作：word1前i-1字符已成功变成word2前j-1个字符，替换word1第i个字符为word2的第j个字符
```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        //.dp[i][j]下标i-1结尾的word1下标j-1结尾的word2，将1转化成2的最小操作数
        vector<vector<int>> dp(word1.size()+1, vector<int>(word2.size()+1));
        //初始化：
        for(int i = 0; i <= word1.size(); i++)dp[i][0] = i;
        for(int j = 0; j <= word2.size(); j++)dp[0][j] = j;
        for(int i = 1; i <= word1.size(); i++){
            for(int j = 1; j <= word2.size(); j++){
                if(word1[i-1] == word2[j-1])dp[i][j] = dp[i-1][j-1];
                else dp[i][j] = min(dp[i-1][j-1] + 1,min(dp[i-1][j], dp[i][j-1]) + 1);
            }
        }
        return dp[word1.size()][word2.size()];
    }
};
```
