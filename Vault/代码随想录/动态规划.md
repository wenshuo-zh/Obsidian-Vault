dp四步解题法：
1. 定义dp数组含义
2. 确定递推公式（推导状态转移方程）
3. dp数组初始化
4. 确定遍历顺序

# 基础题目
## 斐波那契数
```cpp
class Solution {
public:
    int fib(int n) {
        if(n <= 1)return n;
        //1.dp数组含义：下标i的斐波那契数值
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = dp[i - 1] + dp[i - 2];
        //3.初始化
        dp[0] = 0;
        dp[1] = 1;
        //4.遍历顺序：从前向后
        for(int i = 2; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```
## 爬楼梯
```cpp
class Solution {
public:
    int climbStairs(int n) {
        if(n <= 2) return n;
        //1.dp数组含义：到达第i阶的方法数
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = dp[i-1] + dp[i-2]
        //因为到i阶可以从i-1阶走一阶或者从i-2阶走两阶（从i-2走两个一阶被从i-1阶的包含过）
        //3.初始化
        dp[1] = 1;
        dp[2] = 2;
        //4.遍历顺序：从前向后
        for(int i = 3; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```
## 使用最小花费爬楼梯
```cpp
class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
    //1.dp数组含义：到达第i阶台阶顶部的最小花费
    vector<int> dp(cost.size() + 1);
    //2.递推公式：dp[i] = min(dp[i - 1] + cost[i - 1], dp[i - 2] + cost[i - 2])
    //3.初始化
    dp[0] = 0;
    dp[1] = 0;
    //4.顺序：从前向后
    for (int i = 2; i <= cost.size(); i++) {
        dp[i] = min(cost[i - 1] + dp[i - 1], cost[i - 2] + dp[i - 2]);
    }
    return dp[cost.size()];
    }
};
```
## 不同路径
```cpp
class Solution {
public:
    int uniquePaths(int m, int n) {
        //1.dp数组的含义：从起点移动到下标i，j的路径数
        vector<vector<int>> dp(m+1, vector<int>(n+1));
        //2.递推公式：dp[i][j] = dp[i - 1][j] + dp[i][j - 1]
	        //（正上方向下走一步 + 正左侧向右走一步）
        //3.初始化
        for(int i = 1; i <= m; i++)dp[i][1] = 1;
        for(int j = 1; j <= n; j++)dp[1][j] = 1;
        //4.遍历方向：从前向后
        for(int i = 2; i <= m; i++){
            for(int j = 2; j <= n; j++){
                dp[i][j] = dp[i-1][j] + dp[i][j-1];
            }
        }
        return dp[m][n];
    }
};
```
## 不同路径II
踩坑：初始化边界时遇到障碍，只是把当前值设成0，后面的也都走不到了都要为0
```cpp
class Solution {
public:
    int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
        int m = obstacleGrid.size();
        int n = obstacleGrid[0].size();
        //1.dp数组含义：到达ij下标的路径数
        vector<vector<int>> dp(m, vector<int>(n));
        //2. 递推公式：dp[i][j] = dp[i-1][j] + dp[i][j -1]
        //3.初始化
        dp[0][0] = abs(obstacleGrid[0][0] - 1);
        for(int i = 1; i < m; i++){
            if(obstacleGrid[i][0] == 1)dp[i][0] = 0;
            else dp[i][0] = dp[i-1][0];
        }
        for(int j = 1; j < n; j++){
            if(obstacleGrid[0][j] == 1)dp[0][j] = 0;
            else dp[0][j] = dp[0][j-1];
        }
        //4.递推方向：从前向后
        for(int i = 1; i < m; i++){
         for(int j = 1; j < n; j++){
             if(obstacleGrid[i-1][j] == 0)dp[i][j] += dp[i-1][j];
             if(obstacleGrid[i][j-1] == 0)dp[i][j] += dp[i][j-1];
             if(obstacleGrid[i][j] == 1)dp[i][j] = 0;
         }
        }
        return dp[m-1][n-1];
    }
};
```
## 整数拆分
`dp[i]` 表示把整数 `i` 拆成至少两个正整数后的最大乘积。先拆成 `j` 和 `i-j`：`j * (i-j)` 表示剩余的 `i-j` 不再拆，`j * dp[i-j]` 表示继续拆剩余部分；两种情况取最大值。
```cpp
class Solution {
public:
    int integerBreak(int n) {
        //1.dp数组含义：整数i拆成至少两个正整数后的最大乘积
        vector<int> dp(n + 1);
        //2.递推公式：dp[i] = max(j*(i - j), j*dp[i - j])
        //3.初始化
        dp[0] = 0;
        dp[1] = 0; // 1 无法拆成两个正整数，本题不会使用 dp[1]
        dp[2] = 1;
        //4.从前向后遍历
        for(int i = 3;i <= n; i++){
            dp[i] = 0;
            for(int j = 1; j <= i / 2; j++){
                int temp = max(j * (i - j), j * dp[i - j]);
                if(temp > dp[i])dp[i] = temp;
            }
        }
        return dp[n];
    }
};
```
## 不同的二叉搜索树
j是根节点的值，j-1是左子树的节点个数，i-j是右子树的节点个数
```cpp
class Solution {
public:
    int numTrees(int n) {
        //1.dp含义：i个节点组成的二叉搜索树有几种
        vector<int> dp(n+1);
        //2.递推公式：j从1到i dp[i] += dp[j-1] * dp[i-j]
            //j是根节点的值，j-1是左子树的节点个数，i-j是右子树的节点个数
        //3.初始化
        dp[0] = 1;
        //4.从前向后遍历
        for(int i = 1; i <= n; i++){
            for(int j = 1; j <= i; j++){
                dp[i] += dp[j-1] * dp[i-j];
            }
        }
        return dp[n];
    }
};
```
