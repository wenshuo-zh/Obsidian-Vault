
## 哈希表基础
1. 哈希表通过哈希函数把 **key 映射到数组下标**，用空间换取查询速度。
2. 不同 key 得到相同下标时发生**哈希冲突**，常见解决方式：拉链法、线性探测法。
3. 平均情况下，插入、查找、删除的时间复杂度为 O(1)；极端冲突时可能退化为 O(n)。

| 容器              | 适用场景           | 特点              |
| --------------- | -------------- | --------------- |
| `int count[26]` | key 范围固定，如小写字母 | 最快，直接用下标计数      |
| `unordered_set` | 去重、判断某个值是否存在   | 只存 key          |
| `unordered_map` | 统计频率、记录元素位置    | 存 key-value 键值对 |

- `unordered_set`、`unordered_map` 不保证遍历顺序，平均查询 O(1)。
- `set`、`map` 底层是红黑树，元素有序，查询 O(log n)。
- `map[key]` 在 key 不存在时会创建默认值；只查询时优先用 `find()` 或 `count()`。

## 有效的字母异位词
两个字符串中每个字母出现次数相同就是字母异位词。题目只含小写字母，直接用 `int arr[26]` 计数，不需要哈希表。

- 遍历 `s` 加一、遍历 `t` 减一，最后只要有位置不为 0 就不是异位词。
- 下标通过 `s[i] - 'a'` 得到；该做法只适用于字符范围固定的场景。
```cpp
class Solution {
public:
    bool isAnagram(string s, string t) {
        if(s.size() != t.size())return false;
        int arr[26] = {0};
        for(int i = 0; i < s.size(); i++)arr[s[i] - 'a']++;
        for(int i = 0; i < t.size(); i++)arr[t[i] - 'a']--;
        for(int i = 0; i < 26; i++)if(arr[i] != 0)return false;
        return true;
    }
};
```
## 两个数组的交集
交集要求结果**不重复**，用 `unordered_set` 保存第一个数组，再遍历第二个数组判断是否存在。

- 结果也用集合保存，避免第二个数组中有重复元素时重复加入。
- 返回前将集合转为 `vector`；结果顺序不作要求。
```cpp
class Solution {
public:
    vector<int> intersection(vector<int>& nums1, vector<int>& nums2) {
        unordered_set<int> set;
        unordered_set<int> ans;
        for(auto it : nums1){
            set.insert(it);
        }
        for(auto it : nums2){
            if(set.count(it))ans.insert(it);
        }
        return vector<int>(ans.begin(), ans.end());
    }
};
```
## 两数之和
用哈希表记录已经遍历过的数及其下标：`key` 是元素值，`value` 是下标。

- 先查找 `target - nums[i]` 是否出现过，再记录当前元素，避免同一个元素被使用两次。
- 找到补数后直接返回两个下标，时间复杂度 O(n)，空间复杂度 O(n)。
```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        unordered_map<int,int>map;
        for(int i = 0; i< nums.size(); i++){
            auto it = map.find(target - nums[i]);
            if(it != map.end())return{it->second,i};
            map[nums[i]] = i;
        }
        return {};
    }
};
```
## 快乐数
单次计算各位平方和时，`n /= 10` 会遍历数字的每一位，时间复杂度为 O(logn)。平方和会很快进入有限范围，因此题目给定 `int` 范围内，总时间复杂度和空间复杂度可看作 O(1)。
```cpp
class Solution {
public:
    bool isHappy(int n) {
        unordered_set<int>set;
        while(1){
            int sum = 0;
            while(n != 0){
                int temp = n % 10;
                sum += temp * temp;
                n = n / 10;
            }
            n = sum;
            if(sum == 1)return true;
            auto it = set.find(sum);
            if(it != set.end())return false;
            set.insert(sum);
        }
    }
};
```
## 四数相加II
分组哈希：时间复杂度O(n^2)，空间复杂度O(n^2)
```cpp
class Solution {
public:
    int fourSumCount(vector<int>& nums1, vector<int>& nums2, vector<int>& nums3, vector<int>& nums4) {
        unordered_map<int,int> mp1, mp2;
        for(int i  = 0; i < nums1.size(); i++){
            for(int j = 0;j < nums2.size(); j++){
                int sum = nums1[i] + nums2[j];
                mp1[sum]++;
            }
        }
        for(int i  = 0; i < nums3.size(); i++){
            for(int j = 0;j < nums4.size(); j++){
                int sum = nums3[i] + nums4[j];
                mp2[sum]++;
            }
        }
        int ans = 0;
        for(auto it : mp1){
            auto it2 = mp2.find(-it.first);
            if(it2 != mp2.end()){
                ans = ans + it.second * it2->second;
            }
        }
        return ans;
    }
};
```
只用一个哈希表的优化版本：时间复杂度O(n^2)，空间复杂度O(n^2)
```cpp
class Solution {
public:
    int fourSumCount(vector<int>& nums1, vector<int>& nums2, vector<int>& nums3, vector<int>& nums4) {
        unordered_map<int,int>map;
        for(int a : nums1){
            for(int b : nums2){
                //键存放a+b的值，值存放出现次数
                map[a + b]++;
            }
        }
        int ans = 0;
        for(int c : nums3){
            for(int d : nums4){
                int sum = -(c + d);
                auto it = map.find(sum);
                if(it != map.end()){
                    ans += it->second;
                }
            }
        }
        return ans;
    }
};
```
## 赎金信
哈希map版本
```cpp
class Solution {
public:
    bool canConstruct(string ransomNote, string magazine) {
        unordered_map<char,int>map;
        for(char i : magazine){
            map[i]++;
        }
        for(char i : ransomNote){
            auto it = map.find(i);
            if(it == map.end() || it->second == 0)return false;
            --it->second;
        }
        return true;
    }
};
```
数组版本：
- 哈希表为散列表，相比数组会有额外空间开销。
- 此题字符范围固定为 26 个小写字母，直接用 `int count[26]` 或 `vector<int>(26)` 计数即可。
- 当 key 范围已知且很小时，数组下标就是最快的哈希函数。
```cpp
class Solution {
public:
    bool canConstruct(string ransomNote, string magazine) {
        if (ransomNote.size() > magazine.size())return false;
        vector<int>vec(26);
        for (auto& it : magazine)vec[(int)(it - 'a')]++;
        for (auto& it : ransomNote) {
            vec[(int)(it - 'a')]--;
            if (vec[(int)it - 'a'] < 0)return false;
        }
        return true;
    }
};
```
## 三数之和
题目要求找出所有和为 0 且**不重复**的三元组。
### 哈希法
先排序，固定第一个数 `nums[i]`，再用哈希集合记录已经遍历过的第二个数。遍历 `nums[k]` 时，查找补数 `-(nums[i] + nums[k])` 是否出现过。
- `i` 去重：`nums[i] == nums[i - 1]` 时跳过，避免相同的第一个数。
- 内层相同数字最多保留两个，例如 `[-2, 1, 1]` 中两个 `1` 都不能跳过。
- 找到答案后 `erase(target)`，避免同一个三元组重复加入。
- 时间复杂度 O(n^2)，空间复杂度 O(n)。
```cpp
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>> ans;
        int n = nums.size();
        sort(nums.begin(), nums.end());
        for (int i = 0; i < n; i++) {
            if (nums[i] > 0) break;
            if (i > 0 && nums[i] == nums[i - 1]) continue;
            unordered_set<int> set;
            for (int k = i + 1; k < n; k++) {
                if (k > i + 2 && nums[k] == nums[k - 1] &&
                    nums[k - 1] == nums[k - 2]) continue;
                int target = 0 - (nums[i] + nums[k]);
                auto it = set.find(target);
                if (it != set.end()) {
                    ans.push_back({nums[i], target, nums[k]});
                    set.erase(target);
                } else {
                    set.insert(nums[k]);
                }
            }
        }
        return ans;
    }
};
```

### 双指针法（推荐）
排序后固定 `nums[i]`，在区间 `[i + 1, n - 1]` 使用左右指针。
- 三数和大于 0：`right--`，使和变小。
- 三数和小于 0：`left++`，使和变大。
- 三数和等于 0：记录答案，`left++`、`right--` 后跳过重复值。
- 时间复杂度 O(n^2)，空间复杂度 O(1)（不计返回答案）。
```cpp
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>> ans;
        sort(nums.begin(), nums.end());
        int n = nums.size();
        for (int i = 0; i < n; i++) {
            int left = i + 1;
            int right = n - 1;
            if (nums[i] > 0) break;
            if (i > 0 && nums[i] == nums[i - 1]) continue;
            while (left < right) {
                if (nums[i] + nums[left] + nums[right] == 0) {
                    ans.push_back({nums[i], nums[left], nums[right]});
                    left++;
                    right--;
                    while (left < right && nums[left] == nums[left - 1]) left++;
                    while (left < right && nums[right] == nums[right + 1]) right--;
                } else if (nums[i] + nums[left] + nums[right] > 0) {
                    right--;
                } else {
                    left++;
                }
            }
        }
        return ans;
    }
};
```
## 四数之和
题目要求找出所有和为 `target` 且不重复的四元组。先排序，再固定前两个数 `i、j`，对剩余区间使用左右指针。

- `i`、`j` 分别在各自循环开头去重，避免相同起点产生重复答案。
- 三数和小于 `target` 时 `left++`，大于 `target` 时 `right--`。
- 找到答案后先记录，再移动 `left/right`，最后跳过重复值。
- 当前代码使用 `int` 保存总和；如果数据范围较大，四个 `int` 相加可能溢出，需改用 `long long`，并在加法前用 `static_cast<long long>` 转换。
- 时间复杂度 O(n^3)，空间复杂度 O(1)（不计返回结果）。

```cpp
class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {
        vector<vector<int>> ans;
        int n = nums.size();
        sort(nums.begin(), nums.end());
        for (int i = 0; i < n - 3; i++) {
            if (i > 0 && nums[i] == nums[i - 1]) continue;
            for (int j = i + 1; j < n - 2; j++) {
                if (j > i + 1 && nums[j] == nums[j - 1]) continue;
                int left = j + 1;
                int right = n - 1;
                while (left < right) {
                    int sum = nums[left] + nums[right] + nums[i] + nums[j];
                    if (sum < target) left++;
                    else if (sum > target) right--;
                    else {
                        ans.push_back({nums[i], nums[j], nums[left], nums[right]});
                        left++;
                        right--;
                        while (left < right && nums[left] == nums[left - 1])left++;
                        while (left < right && nums[right] == nums[right+ 1])right--;
                    }
                }
            }
        }
        return ans;
    }
};
```
