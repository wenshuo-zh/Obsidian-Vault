1. 内存特点：链表在内存不是连续分布的，通过指针链接在内存的各个节点
2. 复杂度：插入、删除时间复杂度O(1)，查询时间复杂度O(n)
3. 链表的长度可以是不固定的，并且可以动态增删， 适合数据量不固定，频繁增删，较少查询的场景

## 构造链表（尾插法，头插法）
#### 节点定义
```cpp
struct ListNode{
	int val;
	ListNode *next;
	ListNode():val(0),next(nullptr){}
};
```
#### 尾插法
具体操作过程：创建一个虚头节点（不存放数据，方便操作），定义中间变量temp，初始等于dummyHead。循环链表长度次数：==创建节点p，cin值val，将temp的下一项指向p，p的下一项指空，更新temp = temp->next（即指向p）==，最后返回头节点（虚头节点的下一项）。
```cpp
ListNode* creatList(int n){
	ListNode* dummyHead = new ListNode();
	dummyHead->next = nullptr;
	ListNode* temp = dummyHead;
	while(n--){
		ListNode* p = new ListNode();
		cin >> p->val;
		temp->next = p;
		p->next = nullptr;
		temp = temp->next;
	}
	ListNode* realNode = dummyHead->next;
	delete dummyHead;
	return realNode;
}
```
#### 头插法
p的下一项指向头节点（p成为新的头节点），虚头节点的下一项指向p
```cpp
ListNode* creatList(int n){
	ListNode* dummyHead = new ListNode;
	dummyHead->next = nullptr;
	while(n--){
		ListNode* p = new ListNode;
		cin >> p->val;
		p->next = dummyHead->next;
		dummyHead->next = p;
	}
	ListNode* realHead = dummyHead->next;
	delete dummyHead;
	return realHead;
}
```

## 删除链表元素
利用虚头节点和cur来删除链表元素，注意删除节点后cur->next改变，不能直接指向next，否则会漏掉下一个节点。没删除数据才后移。
```cpp
class Solution {
public:
    ListNode* removeElements(ListNode* head, int val) {
        ListNode* dummyhead = new ListNode;
        dummyhead->next = head;
        ListNode* cur = dummyhead;
        while(cur->next != nullptr) {
            if(cur->next->val == val) cur->next = cur->next->next;
            else cur = cur->next;
        }
        return dummyhead->next;
    }
};
```
递归
```cpp
class Solution {
public:
    ListNode* removeElements(ListNode* head, int val) {
        if(head == nullptr)return head;
        if(head->val == val){
            ListNode* newhead = removeElements(head->next, val);
            return newhead;
        }
        else{
            head->next = removeElements(head->next, val);
            return head;
        }
    }
};
```

        }

    void addAtIndex(int index, int val) {

            index = 0;

            cur = cur->next;

## 反转链表
1. next保存链表剩余部分，防止断链丢失
2. cur->next = dummy->next 把当前节点接到虚头节点和头节点之间
3. 更新新的链表头
4. cur走到下一个节点
时间复杂度O(n)，空间复杂度O(1)
```cpp
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode* cur = head;
        if(cur == nullptr)return cur;
        ListNode* dummyhead = new ListNode();
        while(cur != nullptr){
             ListNode* next = cur->next;
             cur->next = dummyhead->next;
             dummyhead->next = cur;
             cur = next;
        }
        return dummyhead->next;
    }
};
```
遍历链表，复制节点到temp，利用头插法插入到新链表里，实现逆序
时间复杂度O(n)，空间复杂度O(n)
```cpp
class Solution{
	public:
	ListNode* reverseList(ListNode* head){
		ListNode* dummyhead = new ListNode();
		for(ListNode* i = head; i != nullptr; i = i->next){
			ListNode* temp = new ListNode();
			temp->val = i->val;
			temp->next = dummyhead->next;
			dummyhead->next = temp;
		}
		return dummyhead->next;
	}
};
```
双指针法：

```cpp
class Solution{
	public:
	ListNode* reverseList(ListNode* head){
		ListNode* pre = nullptr;
		ListNode* cur = head;
		while(cur != nullptr){
			ListNode* temp = cur->next;
			cur->next = pre;
			pre = cur;
			cur = temp;
		}
		return pre;
	}
};
```

## 删除链表的倒数第N个节点（快慢指针）
利用快慢指针，快指针先走n步，慢指针不动，然后两个指针一起移动，快指针走到nullptr的时候慢指针所在的节点就是倒数第n个，所以开始慢指针要在快指针前一个节点，即慢指针在dummyhead，快指针在head
```cpp
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode* dummyHead = new ListNode;
        dummyHead->next = head;
        ListNode* fast = head;
        ListNode* slow = dummyHead;
        while (n--) {
            fast = fast->next;
        }
        while (fast != nullptr) {
            fast = fast->next;
            slow = slow->next;
        }
        slow->next = slow->next->next;
        return dummyHead->next;
    }
};
```
## 链表相交
先求长度差，再让较长链表先走几步，最后同步遍历比较节点地址
```cpp
class Solution {
public:
    int getLength(ListNode* head){
        int length = 0;
        while(head){
            head = head->next;
            length++;
        }
        return length;            
    }
    
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        int n = getLength(headA) - getLength(headB);
        ListNode* curA = headA;
        ListNode* curB = headB;
        if(n > 0){
            while(n--)curA = curA->next;
        }else{
            while(n++) curB = curB->next;
        }
        while(curA != nullptr){
            if(curA == curB){
                return curA;
            }
            curA = curA->next;
            curB = curB->next;
        }
        return nullptr;
    }
};
```

## 环形链表II
快慢指针：快慢指针第一次相遇说明链表存在环；然后让一个指针从head出发，另一个从相遇位置出发，两者同步遍历，相遇处就是环入口
时间复杂度O(n)，空间复杂度O(1)
```cpp
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        ListNode *fast = head, *slow = head;
        while(fast != nullptr && fast->next != nullptr){
            slow = slow->next;
            fast = fast->next->next;
            if(fast == slow){
                ListNode* ans = head;
                while(slow != ans){
                    ans = ans->next;
                    slow = slow->next;
                }
                return ans;
            }
        }
        return nullptr;
    }
};
```
