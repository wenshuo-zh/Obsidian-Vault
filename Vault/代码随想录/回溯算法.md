```cpp
class Solution {
public:
    vector<vector<int>> res;
    vector<int> path;
    void backtracking(int n, int k, int startIndex){
        if(path.size() == k){
            res.push_back(path);
            return;
        }
        for(int i = startIndex; i <= n; i++){
            path.push_back(i);
            backtracking(n, k, i+1);
            path.pop_back();
        }
    }

        backtracking(n, k, 1);
        return res;
    }
};
```
**剪枝优化：** 当待选数量（k - path.size()）大于包含i的剩余数量（n - i + 1）的部分就剪枝掉，不迭代，即 👇
```cpp
if(k-path.size() > (n - i + 1)) break;
```
写到循环条件里即👇
```cpp
for(int i = startIndex; i <= n + 1 - (k - path.size()); i++){}
```

## 电话号码的字母组合
抽象成树结构，digits位数作深度，digits当前位映射的map字符串长度作宽度
digitsIndex表示当前处理到digits的第几个数字
mapIndex表示当前要处理的数字，注意要字符型转整型
```cpp
class Solution {
public:
    const string map[10]={
        "",
        "",
        "abc",
        "def",
        "ghi",
        "jkl",
        "mno",
        "pqrs",
        "tuv",
        "wxyz",
    };
    vector<string> res;
    string str;
    void backtracking(string digits, int digitsIndex){
        if(str.size() == digits.size()){
            res.push_back(str);
            return;
        }
        int mapIndex = digits[digitsIndex] - '0';
        for(int i = 0; i < map[mapIndex].size(); i++){
            str += map[mapIndex][i];
            backtracking(digits, digitsIndex + 1);
            str.pop_back();
        }
    }
    vector<string> letterCombinations(string digits) {
        if(digits.empty())return res;
        backtracking(digits, 0);
        return res;
    }
};
```

## 组合总和II
树层重复：比如说第一个1和7组合，第二个1还和7组合，重复
解决方法：
`if(i > startIndex && candidates[i] == candidates[i-1]) continue;`
先对数组排序、在遍历数层的时候相同元素一定挨着，遇到相同的跳过就行
```cpp
class Solution {
public:
    vector<vector<int>> res;
    vector<int> vec;
    void backtracking(vector<int>& candidates, int target, int startIndex){
        if(target == 0){
            res.push_back(vec);
            return;
        }
        if(target < 0)return;
        for(int i = startIndex; i < candidates.size(); i++){
            if(i > startIndex && candidates[i] == candidates[i - 1])continue;
            vec.push_back(candidates[i]);
            backtracking(candidates, target - candidates[i], i + 1);
            vec.pop_back();
        }
    }
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        sort(candidates.begin(), candidates.end());
        backtracking(candidates, target, 0);
        return res;
    }
};
```
**剪切优化：** 同上一题，已经排过序了，直接在循环条件加上candidates\[i] <= target
```cpp

```
`if(target < 0)return;`不会出现这种情况了，删掉这行就行

## 分割回文串
核心要点理解：
	for 循环的 i 用来**不断拉长当前这一刀的终点**，每选定一个 i，就把**i+1 后面剩下的所有字符交给递归深度优先搜索**； 等这一刀对应的所有分割方案全部搜索、回溯完毕，再回来 for 循环，i++，尝试更长的一刀
- ==`s.substr()`==，作用将字符串分割成子字符串，第一个参数传起始下标、第二个参数传总分割位数
```cpp
class Solution {
public:
    bool isp(const string& s, int startIndex, int i){
        if(s.empty())return false;
        while(startIndex <= i){
            if(s[startIndex++] != s[i--])return false;  
        }
        return true;
    }
    vector<vector<string>> res;
    vector<string> path;
    void backtracking(const string& s, int startIndex){
        if(startIndex >= s.size()){
            res.push_back(path);
            return;
        }
        for(int i = startIndex; i < s.size(); i++){
            //截取[startIndex, i]部分的子串
            if(isp(s, startIndex, i)){
                string str = s.substr(startIndex, i - startIndex + 1);
                path.push_back(str);
                backtracking(s, i + 1);
                path.pop_back();
            }
        }
    }
    vector<vector<string>> partition(string s) {
        backtracking(s, 0);
        return res;
    }
};
```

## 复原IP地址
和上一题对比，相当于只能切三刀，直接在原字符串切处插入`'.'` ，回溯删除`'.'`即可，借助pointnums来记录已经切了几刀
- ==`s.insert()`== 在字符串指定位置插入字符、子串
	- `s.insert(pos, str)`：在下标pos，插入字符串str
	- `s.insert(pos, n, ch)`：在下标pos，插入n个字符ch
- ==`s.erase()`== 删除字符串部分字符
	- `s.erase(pos, length)`：下标pos开始，删掉后面length个字符
	- `s.erase(迭代器)`：删除单个字符
	- `s.erase(迭代器左, 迭代器右)`：删除左闭右开区间
```cpp
class Solution {
public:
    vector<string> res;
    bool isValid(const string& s, int i, int j){
        if(i > j)return false;
        if(j > i && s[i] == '0')return false;
        int num = 0;
        for(int index = i; index <= j; index++){
            if(s[index] > '9' || s[index] < '0')return false;
            num = num * 10 + (s[index] - '0');
            if(num > 255)return false;
        }
        return true;
    }
    void backtracking(string &s, int startIndex, int pointNums){
        if(pointNums == 3){
            if(isValid(s, startIndex, s.size() - 1))res.push_back(s);
            return;
        }
        for(int i = startIndex; i < s.size(); i++){
            if(isValid(s, startIndex, i)){
                s.insert(s.begin() + i + 1, '.');
                backtracking(s, i + 2, pointNums + 1);
                s.erase(s.begin() + i + 1);
            }
            else break;
        }
    }
    vector<string> restoreIpAddresses(string s) {
        backtracking(s, 0, 0);
        return res;
    }
};
```
