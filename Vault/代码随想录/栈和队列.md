
## 用栈实现队列
用两个栈，一个in栈一个out栈，in负责入队out负责出队，out为空时直接把in中元素倒入out，就能实现队列的先入先出
```cpp
class MyQueue {
public:
    stack<int>stIn;
    stack<int>stOut;

    MyQueue() {

    }

    void push(int x) {
        stIn.push(x);
    }

    int pop() {
        if(stOut.empty()){
            while(!stIn.empty()){
                stOut.push(stIn.top());
                stIn.pop();
            }
        }
        int ans = stOut.top();
        stOut.pop();
        return ans;
    }

    int peek() {
        if(stOut.empty()){
            while(!stIn.empty()){
                stOut.push(stIn.top());
                stIn.pop();
            }
        }
        return stOut.top();
    }

    bool empty() {
        return stOut.empty() && stIn.empty();
    }
};
```
时间复杂度：`push` 为 `O(1)`；`pop` 和 `peek` 单次最坏为 `O(n)`，均摊为 `O(1)`。
空间复杂度：`O(n)`。

## 用队列实现栈

```cpp
class MyStack {
public:
    queue<int>q1;
    queue<int>q2;
    MyStack() {

    }
    void push(int x) {
        q1.push(x);
    }

    int pop() {
        while(q1.size() > 1){
            q2.push(q1.front());
            q1.pop();
        }
        int temp = q1.front();
        q1.pop();
        q1 = q2;
        while(!q2.empty())q2.pop();
        return temp;
    }

    int top() {
        while(q1.size() > 1){
            q2.push(q1.front());
            q1.pop();
        }
        q2.push(q1.front());
        int temp = q1.front();
        q1 = q2;
        while(!q2.empty())q2.pop();
        return temp;
    }

    bool empty() {
        return q1.empty();
    }
};
```
时间复杂度：`push` 为 `O(1)`；`pop` 和 `top` 为 `O(n)`；`empty` 为 `O(1)`。
空间复杂度：`O(n)`。

## 有效的括号
遍历字符串，如果是左括号就将对应的右括号入栈，如果是右括号就去看栈顶是不是一样的右括号，是就出栈，不是就无效
```cpp
class Solution {
public:
    bool isValid(string s) {
        stack<char>st;
        for(char ch : s){
            if(ch == '(')st.push(')');
            else if(ch == '{')st.push('}');
            else if(ch == '[')st.push(']');
            else{
                if(st.empty() || st.top() != ch)return false;
                st.pop();
            }
        }
        return st.empty();
    }
};
```
时间复杂度：`O(n)`，每个括号最多入栈、出栈一次。
空间复杂度：`O(n)`，最坏情况下所有字符都是左括号。

遇到右括号时，要先判断栈是否为空，再访问栈顶；遍历结束后栈必须为空，才说明括号完全匹配。
## 删除字符串中所有相邻重复项
```cpp
class Solution {
public:
    string removeDuplicates(string s) {
        stack<char> st;
        for(char ch : s){
            if(st.empty())st.push(ch);
            else {
                if(ch == st.top())st.pop();
                else st.push(ch);
            }
        }
        string ans;
        while(!st.empty()){
            ans += st.top();
            st.pop();
        }
        reverse(ans.begin(),ans.end());
        return ans;
    }
};
```
时间复杂度：`O(n)`，每个字符最多入栈、出栈一次，最后反转一次。
空间复杂度：`O(n)`。

## 逆波兰表达式求值
```cpp
class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        stack<int> st;
        for(int i = 0;i < tokens.size(); i++){
            if(tokens[i] == "+" || tokens[i] == "-" || tokens[i] == "*" || tokens[i] == "/"){
                int nums1 = st.top();
                st.pop();
                int nums2 = st.top();
                st.pop();
                if(tokens[i] == "+")st.push(nums2 + nums1);
                else if(tokens[i] == "-")st.push(nums2 - nums1);
                else if(tokens[i] == "*")st.push(nums2 * nums1);
                else st.push(nums2 / nums1);
            }
            else{
                int temp = stoi(tokens[i]);
                st.push(temp);
            }
        }
        return st.top();
    }
};
```
时间复杂度：`O(n)`，每个 token 只入栈或出栈一次。
空间复杂度：`O(n)`。

## 滑动窗口最大值（单调队列）

队列要存下标而不是数值。
单调队列：对应数值大的下标在队头，对应数值小的下标在队尾。
窗口右滑一个数：
1. 先检查队头，如果下标滑出窗口，弹出队头（队列存数值就没法知道队头有没有滑出）
2. 从队尾删掉所有比新数小的下标，保持单调队列
3. 最后把新下标加入队尾
```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        deque<int> dq;
        vector<int> ans;
        for(int i = 0; i < nums.size(); i++){
            if(!dq.empty() && dq.front() <= i - k)dq.pop_front();
            while(!dq.empty() && nums[dq.back()] <= nums[i])dq.pop_back();
            dq.push_back(i);
            if(i >= k - 1){
                ans.push_back(nums[dq.front()]);
            }
        }    
        return ans;
    }
};
```
时间复杂度：`O(n)`，每个下标最多入队和出队一次。
空间复杂度：`O(k)`。

## 前K个高频元素（优先队列、小顶堆）
```cpp
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        //手写仿函数cmp，用作比较器
        //sort()：cmp(a,b)返回true，a放在b前面
        //priority_queue：cmp(a,b)返回true，a靠近堆底，b靠近堆顶
        class mycomparison{
            public:
            bool operator()(const pair<int,int>& lhs,const pair<int,int>& rhs){
                return lhs.second > rhs.second;
            }
        };
        unordered_map<int,int> mp;//key存数值，val存频率
        for(int i = 0; i < nums.size(); i++){
            mp[nums[i]]++;
        }

        //优先级队列（小顶堆）：<元素类型，底层容器，比较器>
        priority_queue<pair<int,int>,vector<pair<int,int>>,mycomparison> minHeap;
        for(auto &it : mp){
            minHeap.push(it);
            if(minHeap.size() > k)minHeap.pop();
        }
        vector<int>ans;
        while(!minHeap.empty()){
            ans.push_back(minHeap.top().first);
            minHeap.pop();
        }
        return ans;
    }
};
```
思路：先用哈希表统计每个数字的出现频率，再用小顶堆维护频率最高的 `k` 个数字。
	小顶堆的堆顶是当前频率最小的数字。当堆的大小超过 `k` 时，弹出堆顶，保证堆中始终保留前 `k` 个高频元素。
时间复杂度：`O(n + m log k)`，其中 `m` 是不同元素的数量。
空间复杂度：`O(m + k)`。
