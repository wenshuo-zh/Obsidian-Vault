## 对称二叉树
**递归法：**
```cpp
class Solution {
public:
    bool compare(TreeNode* left, TreeNode* right){
        if(left == nullptr && right == nullptr)return true;
        else if(left == nullptr && right != nullptr)return false;
        else if(left != nullptr && right == nullptr)return false;
        else if(left->val != right->val)return false;
        bool out = compare(left->left, right->right);
        bool in = compare(left->right, right->left);
        return out && in;
    }
    bool isSymmetric(TreeNode* root) {
        if(root == nullptr)return true;
        return compare(root->left, root->right);
    }
};
```
**迭代法：** 
```cpp
class Solution {
public:
    bool isSymmetric(TreeNode* root) {
        queue<TreeNode*> que;
        if(root == nullptr)return true;
        que.push(root->left);
        que.push(root->right);
        while(!que.empty()){
            TreeNode* leftNode = que.front();
            que.pop();
            TreeNode* rightNode = que.front();
            que.pop();
            if(leftNode == nullptr && rightNode == nullptr)continue;
            if(leftNode == nullptr || rightNode == nullptr || leftNode->val != rightNode->val)return false;  
            que.push(leftNode->left);
            que.push(rightNode->right);
            que.push(leftNode->right);
            que.push(rightNode->left);          
        }
        return true;
    }
};
```
## 完全二叉树的节点个数
**递归法：** 
用最左和最右边节点深度对比是不是满二叉树，是满二叉树返回公示，不是满二叉树差分成左右子树继续递归，再加上根节点返回
时间复杂度为O(logn^2)
```cpp
class Solution {
public:
    int countNodes(TreeNode* root) {
        if(root == nullptr)return 0;
        TreeNode* leftNode = root->left;
        TreeNode* rightNode = root->right;
        int leftdepth = 0, rightdepth = 0;
        while(leftNode != nullptr){
            leftNode = leftNode->left;
            leftdepth++;
        }
        while(rightNode != nullptr){
            rightNode = rightNode->right;
            rightdepth++;
        }
        if(leftdepth == rightdepth)return pow(2, leftdepth+1) - 1;
        return countNodes(root->left) + countNodes(root->right) + 1;
    }
};
```
迭代法暴力遍历二叉树，时间复杂度O(n)
## 平衡二叉树
递归法：gethight后序遍历，找左右子树的高度，如果返回-1则向上传递-1，表示不是平衡二叉树，否则返回该子树的高度
```cpp
class Solution {
public:
    int getHeight(TreeNode* node){
        if(node == nullptr)return 0;
        int leftDepth = getHeight(node->left);
        if(leftDepth == -1)return -1;
        int rightDepth = getHeight(node->right);
        if(rightDepth == -1)return -1;
        if(abs(leftDepth - rightDepth) > 1)return -1;
        else return max(leftDepth, rightDepth) + 1;        
    }
    bool isBalanced(TreeNode* root) {
        if(getHeight(root) == -1)return false;
        else return true;
    }
};
```
## 二叉树的所有路径
个人理解getpath有两种return：第一种 return 是走到叶子节点主动 return，代表一条完整路径收集完成；第二种是节点的左右子树全部遍历完成，走到函数末尾隐式 return，回到上一层进行回溯
```cpp
class Solution {
public:
    void getPath(TreeNode* node, vector<int> &path, vector<string> &res){
        path.push_back(node->val);
        if(node->left == nullptr && node->right == nullptr){
            string temp;
            int n = path.size();
            for(int i = 0; i < n - 1; i++){
                temp += to_string(path[i]);
                temp += "->";
            }
            temp += to_string(path[n - 1]);
            res.push_back(temp);
            return;
        }
        if(node->left != nullptr){
            getPath(node->left, path, res);
            path.pop_back();
        }
        if(node->right != nullptr){
            getPath(node->right, path, res);
            path.pop_back();
        }
    }
    vector<string> binaryTreePaths(TreeNode* root) {
        vector<string> res;
        vector<int> path;
        getPath(root, path, res);
        return res;
    }
};
```
## 左叶子之和
```cpp
class Solution {    
public:
    int sumOfLeftLeaves(TreeNode* root) {
        if(root == nullptr)return 0;
        int leftval = sumOfLeftLeaves(root->left);
        if(root->left != nullptr &&root->left->left ==nullptr && root->left->right == nullptr){//找到左叶子
            leftval = root->left->val;
        }
        int rightval = sumOfLeftLeaves(root->right);//右侧子树里面的左叶子之和
        return leftval + rightval;//多个左叶子的情况，这个和被赋值到leftval(便于个人理解)
    }
};
```
