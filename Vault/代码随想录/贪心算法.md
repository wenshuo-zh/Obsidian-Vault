#贪心
## 分发饼干
从小到大遍历饼干，和当前胃口最小的孩子对比，如果能吃饱就下一个饼干和孩子，如果吃不饱就把这个饼干扔掉换下一个饼干
局部贪心👇
	一块饼干，如果连**当前剩下胃口最小**的孩子都喂不饱，那剩下所有孩子胃口都比他更大，这块饼干谁都喂不了 → **直接丢弃这块饼干**
```cpp
class Solution {
public:
    int findContentChildren(vector<int>& g, vector<int>& s) {
        sort(g.begin(), g.end());
        sort(s.begin(), s.end());
        int index = 0;
        for(int i = 0; i < s.size(); i++){
            if(index < g.size() && g[index] <= s[i]){
                index++;
            }
        }
        return index;
    }
};
```
## 摆动序列
数据差作为坡度，遇到三个点是连续单调的就不符合，所以做删除，保留所有峰值，即是最大摆动序列
局部贪心👇
	单调区间**直接取端点作为拐点**，不尝试中间点；每一步局部最优，最终得到全局最长。
```cpp
class Solution {
public:
    int wiggleMaxLength(vector<int>& nums) {
        if(nums.size() <= 1)return nums.size();
        int cur = 0;
        int pre = 0;
        int res = 1;
        for(int i = 0; i < nums.size() - 1; i++){
            cur = nums[i + 1] - nums[i];
            if(cur == 0)continue;
            if((pre <= 0 && cur > 0) || pre >= 0 && cur < 0){
                res++;
            }
            pre = cur;
        }
        return res;
    }
};
```

## 跳跃游戏
只维护覆盖范围，只要覆盖到最远下标就可以跳过来
```cpp
class Solution {
public:
    bool canJump(vector<int>& nums) {
        int cover = 0;
        if (nums.size() == 1) return true;
        for(int i = 0; i < nums.size(); i++){
            if(cover >= i) cover = max(cover, i + nums[i]);
            if(cover >= nums.size() - 1)return true;
        }
        return false;
    }
};
```
## 跳跃游戏II
cur为跳跃res次后能到达的最远下标，next为再跳一次能走到的最远下标
i\==cur时走到边界，res++，cur更新为next
```cpp
class Solution {
public:
    int jump(vector<int>& nums) {
        if(nums.size() == 1)return 0;
        int cur = 0;
        int res = 0;
        int next = 0;
        for(int i = 0; i < nums.size(); i++){
            next = max(next, nums[i] + i);
            if(i == cur){
                res++;
                cur = next;
                if(next >= nums.size() - 1)break;
            }
        }
        return res;
    }
};
```
## k次取反后的最大化数组和
```cpp
class Solution {
public:
    int largestSumAfterKNegations(vector<int>& nums, int k) {
        int res = 0;
        sort(nums.begin(), nums.end());
        int minAbs = abs(nums[0]);
        for(int i = 0;i < nums.size(); i++){
            if(nums[i] < 0 && k > 0){
                nums[i] *= -1;
                k--;
            }
            minAbs = min(minAbs, abs(nums[i]));
        }
        if(k % 2 == 1)res -= 2 * minAbs;
        for(int it : nums)res += it;
        return res;
    }
};
```
## 加油站
### 前缀和解法
#前缀和
第一个for跑完，我们得到`min`为所有前缀和的最小值，`sum`为全部vec总和
	如果sum<0了，代表总油量小于总路程，肯定跑不完
	min>=0了，代表从起点0开始，所有小路程都小于油量，所有下标0为答案
	min<0了，代表不能起始从0开始，我们尝试换起点
第二个for换起点，因为是环形路线，起点换到0的前一项也就是下标n-1，min += vec\[i]就是模拟新起点的最小前缀和，这个大于0了就代表着全程油箱不会为负数，返回答案
```cpp
class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int n = gas.size();
        vector<int> vec(n);
        int sum = 0;
        int min = INT_MAX;
        for(int i = 0; i < n; i++){
            vec[i] = gas[i] - cost[i];
            sum += vec[i];
            if(sum < min){
                min = sum;
            }
        }
        if(sum < 0)return -1;
        if(min >= 0)return 0;
        for(int i = n - 1; i >= 0; i--){
            min += vec[i];
            if(min >= 0)return i;
        }
        return -1;
    }
};
```
### 贪心解法

```cpp
class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int cursum = 0;
        int start = 0;
        int totalsum = 0;
        for(int i = 0; i < gas.size(); i++){
            cursum += gas[i] - cost[i];
            totalsum += gas[i] - cost[i];
            if(cursum < 0){
                start = i + 1;
                cursum = 0;
            }
        }
        if(totalsum < 0)return -1;
        return start;
    }
};
```
- **前缀和版本：先算完全部数据，数学上找到切割点；属于「先全部算完，再找答案」** 
- **贪心版本：边走边试，遇到油不够直接抛弃前面一段，立刻换起点；属于「遍历的同时动态找答案」**
## 分发糖果
两次局部贪心：从左到右遍历，保证右边比左边大；在从右到左遍历，保证左边比右边大
	踩坑点：第二次遍历（从右往左）不要直接 `child[i-1] = child[i]+1`。 当左边分数 > 右边分数时，右边 + 1 得到的糖果数，**有可能比左边已有的糖果数更小**。直接赋值会覆盖第一次遍历已经合法的结果，破坏上坡的约束。 所以要用 `child[i-1] = max(child[i-1], child[i]+1)`，保留原来合法的值，只有需要更大时才更新。
```cpp
class Solution {
public:
    int candy(vector<int>& ratings) {
        int n = ratings.size();
        int res = 0;
        vector<int> child(n, 1);
        //从左向右遍历，如果右边的比左边大就加一
        for(int i = 0; i < n - 1; i++){
            if(ratings[i] < ratings[i+1])child[i+1] = child[i] + 1;
        }
        //左边比右边大,取下标i-1的糖和当前i的糖加一中最大的
        for(int i = n - 1; i > 0; i--){
           if(ratings[i-1] > ratings[i])child[i-1] = max(child[i-1], child[i] + 1);
        }
        for(int i : child)res += i;
        return res;
    }
};
```
## 柠檬水找零
贪心思想：找零20元时优先找`一张10一张5`而不是`三张5`，因为10块钱只有找零20能用上，5块钱都能用上
```cpp
class Solution {
public:
    bool lemonadeChange(vector<int>& bills) {
        int five = 0, ten = 0;
        for(int i = 0; i < bills.size(); i++){
            if(bills[i] == 5)five++;
            if(bills[i] == 10){
                if(five == 0)return false;
                five--;
                ten++;
            }
            if(bills[i] == 20){
                if(five == 0 || (ten == 0 && five <= 2))return false;
                if(ten > 0){
                    ten--;
                    five--;
                }
                else five -= 3;
            }
        }
        return true;
    }
};
```
## 根据身高重建队列
贪心思想：先放高个，达到局部最优，让已在队列的人满足条件。因为条件k与比自己个子低的人无关，所以后放入小个子不会影响局部最优
### 使用vector
整体时间复杂度O(n^2)
```cpp
class Solution {
public:
//按第0个元素降序排列，第0个元素相同则按第1个元素升序排列
    static bool cmp(vector<int>& a, vector<int>& b){
        if(a[0] == b[0])return a[1] < b[1];
        return a[0] > b[0];
    }
    vector<vector<int>> reconstructQueue(vector<vector<int>>& people) {
        sort(people.begin(), people.end(), cmp);
        int n = people.size();
        vector<vector<int>> queue;
        for(int i = 0; i < n; i++){
            int k = people[i][1];//在下标为i同学前面有k个身高大于等于他的同学
            queue.insert(queue.begin() + k, people[i]);
        }
        return queue;
    }
};
```
### 使用链表
```cpp
class Solution {
public:
//按第0个元素降序排列，第0个元素相同则按第1个元素升序排列
    static bool cmp(vector<int>& a, vector<int>& b){
        if(a[0] == b[0])return a[1] < b[1];
        return a[0] > b[0];
    }
    vector<vector<int>> reconstructQueue(vector<vector<int>>& people) {
        sort(people.begin(), people.end(), cmp);
        int n = people.size();
        list<vector<int>> lst;
        for(int i = 0; i < n; i++){
            int k = people[i][1];//在下标为i同学前面有k个比他高的同学
            auto it = lst.begin();
            while(k--)it++;
            lst.insert(it, people[i]);
        }
        vector<vector<int>> queue(lst.begin(), lst.end());
        return queue;
    }
};
```
