## 二分查找

二分查找的核心是**边界问题**——区间如何定义，决定了循环条件和边界更新方式。

#### 左闭右开 `[left, right)`

有效下标满足：`left <= index < right`

```cpp
// 时间复杂度 O(log n)，空间复杂度 O(1)
class Solution {
public:
    int search(vector<int>& nums, int target) {
        int left = 0;
        int right = nums.size();
        // left == right 没有意义，即 [left, right) 区间为空，所以循环条件不包含 =
        while (left < right) {
            int mid = (left + right) / 2;
            if (nums[mid] == target) return mid;
            // mid 比对完毕，下一轮区间里不能包含 mid
            else if (nums[mid] > target) {
                right = mid;      // 新区间 [left, mid)
            } else {
                left = mid + 1;   // 新区间 [mid + 1, right)
            }
        }
        return -1;
    }
};
```

#### 左闭右闭 `[left, right]`

```cpp
// 时间复杂度 O(log n)，空间复杂度 O(1)
class Solution {
public:
    int search(vector<int>& nums, int target) {
        int left = 0, right = nums.size() - 1;
        // left == right 有意义，即 [left, right] 内有值，循环继续执行
        while (left <= right) {
            int mid = (left + right) / 2;
            if (nums[mid] == target) return mid;
            // 新区间不要带上 mid，因为 mid 已经比对
            else if (nums[mid] > target) right = mid - 1;  // [left, mid - 1]
            else left = mid + 1;                           // [mid + 1, right]
        }
        return -1;
    }
};
```

## 删除元素（快慢指针）

双指针（快慢指针法）：

- **fast**：快指针遍历数组，找到要保留的元素
- **slow**：慢指针从头开始，接收快指针找到的有效元素（把要保留的元素覆盖到数组前面）

```cpp
// 时间复杂度 O(n)，空间复杂度 O(1)
class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        int n = nums.size();
        int slow = 0;
        for (int fast = 0; fast < n; fast++) {
            if (nums[fast] != val) {
                nums[slow] = nums[fast];
                slow++;
            }
        }
        return slow;
    }
};
```

## 有序数组的平方（左右指针）

因为新数组的最大值一定在最左或最右，所以用左右指针相向前进遍历原数组，把最大值写到新数组的末尾。

```cpp
// 时间复杂度 O(n)，空间复杂度 O(1)
class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        int n = nums.size();
        vector<int> ans(n);
        int first_i = 0, end_i = n - 1;
        for (int i = n - 1; first_i <= end_i; i--) {
            if (nums[first_i] * nums[first_i] > nums[end_i] * nums[end_i]) {
                ans[i] = nums[first_i] * nums[first_i];
                first_i++;
            }
            else {
                ans[i] = nums[end_i] * nums[end_i];
                end_i--;
            }
        }
        return ans;
    }
};
```

## 长度最小的子数组（滑动窗口）

滑动窗口：左右指针之间的窗口即为目标子数组。起始左右指针都在 `0`；右指针向右扩大窗口，把元素累加到 `sum`，直到 `sum >= target`；再用 `while` 缩小窗口（左指针右移），不断更新全局最小长度。因为数组中每个元素只进、出窗口一次，时间复杂度 O(n)。

```cpp
// 时间复杂度 O(n)，空间复杂度 O(1)
class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int n = nums.size();
        int ans = n + 1;
        int sum = 0;
        int left = 0;
        for (int right = 0; right < n; right++) {
            sum += nums[right];
            while (sum >= target) {
                int len = right - left + 1;
                ans = min(len, ans);
                sum -= nums[left];
                left++;
            }
        }
        return ans == n + 1 ? 0 : ans;
        /*等价于：
            if(ans == n + 1)return 0;
            else return ans;
        */
    }
};
```

### 题解
#### 填边
上->右->下->左 的顺序填边，填完一条边就收缩边界
```cpp
//收缩上下左右 四条边界
class Solution {
public:
    vector<vector<int>> generateMatrix(int n) {
        vector<vector<int>>nums(n, vector<int>(n));
        //边界左闭右开   上->右->下->左 
        int up = 0, down = n, left = 0, right = n;
        int val = 1;
        while (val <= n * n) {
            //固定上边界：行up，遍历[left , right)，结束up++
            for (int j = left; j < right; j++) {
                nums[up][j] = val;
                val++;
            }
            up++;
            //固定右边界：列right - 1，遍历[up , down)，结束right--
            for (int i = up; i < down; i++) {
                nums[i][right - 1] = val;
                val++;
            }
            right--;
            //固定下边界：行down - 1，反向遍历[left , right)，结束down--
            for (int j = right - 1; j >= left; j--) {
                nums[down - 1][j] = val;
                val++;
            }
            down--;
            //固定左边界：列left，反向遍历[up , down)，结束left++
            for (int i = down - 1; i >= up; i--) {
                nums[i][left] = val;
                val++;
            }
            left++;
        }
        return nums;
    }
};
```
#### 缩圈
loop控制圈数，走完一圈在收缩边界，中心点单独赋值
```cpp
//缩圈
class Solution {
public:
    vector<vector<int>> generateMatrix(int n) {
        vector<vector<int>>vec(n, vector<int>(n));
        int loop = n / 2;//圈数
        int val = 1;
        int startX = 0, startY = 0;
        int offset = 1;//控制每条边的遍历长度
        while (loop--) {
            int i = startX;
            int j = startY;
            for (; j < n - offset; j++)vec[i][j] = val++;
            for (; i < n - offset; i++)vec[i][j] = val++;
            //此时i j已经是n-offset,循环中不用再写
            for (; j > startY; j--)vec[i][j] = val++;
            for (;i>startX; i--)vec[i][j] = val++;
            //走完一圈以后边界向内缩小
            startX++;
            startY++;
            offset++;
        }
        //填充中心点
        if (n % 2 != 0) {
            vec[n / 2][n / 2] = val;
        }
        return vec;
    }
};
```

#### 开发商购买土地（前缀和）
abs()函数找绝对值
注意边界
踩坑点：不能在构造前缀和的同时就去进行分割找最小值，因为sum[n-1]还没有赋值。
```cpp
#include<iostream>
#include<vector>
#include<climits>
using namespace std;
int main() {
	int n, m;
	cin >> n >> m;
	//vector<vector<int>>vec(n, vector<int>(m, 0));
	vector<int>row(n, 0), sumRow(n, 0);
	vector<int>col(m, 0), sumCol(m, 0);
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < m; j++) {
			int temp;
			cin >> temp;
			row[i] += temp;
			col[j] += temp;
		}
	}
	sumRow[0] = row[0];
	sumCol[0] = col[0];
	int ans = INT_MAX;
	for(int i = 1;i< n;i++)sumRow[i] = sumRow[i - 1] + row[i];
	for (int i = 0; i < n - 1; i++) {
		int temp = (sumRow[n - 1] - sumRow[i]) - sumRow[i];
		//abs求绝对值
		ans = min(ans, abs(temp));
	}

	for(int j = 1;j<m;j++)sumCol[j] = sumCol[j - 1] + col[j];
	for (int j = 0; j < m - 1; j++) {
		int temp = (sumCol[m - 1] - sumCol[j]) - sumCol[j];
		ans = min(ans, abs(temp));
	}
	cout << ans;

	return 0;
}
```
