# 基础思想+模板
核心作用：维护集合连通关系，适合无向图；关系是双向等价：u 和 v 连通 ⇔ v 和 u 连通，不能直接表达单向有向图的可达性。解决冗余连接II时，并查集只用于检测忽略方向后的环，入度为2的情况需额外处理。
数组含义：`father[x]` 代表 x 的**直接父节点**，不一定是根。
判断连通规则：**两个节点根节点相同，就属于同一个集合**。
### 模板代码
```cpp
int n = 1005; // n根据题目中节点数量而定，一般比节点数量大一点就好
vector<int> father = vector<int> (n, 0); // C++里的一种数组结构
// 并查集初始化
void init() {
    for (int i = 0; i < n; i++) {
        father[i] = i;
    }
}
// 并查集里寻根的过程
int find(int u) {
	if (u == father[u])return u;
	else {
		father[u] = find(father[u]);
		return father[u];
    }
}
// 判断 u 和 v是否找到同一个根
bool isSame(int u, int v) {
    u = find(u);
    v = find(v);
    return u == v;
}
// 将v->u 这条边加入并查集
void join(int u, int v) {
    u = find(u); // 寻找u的根
    v = find(v); // 寻找v的根
    if (u == v) return ; // 如果发现根相同，则说明在一个集合，不用两个节点相连直接返回
    father[v] = u;
}
```
# 卡码网题目
### 107寻找存在的路径
```cpp
#include<iostream>
#include<vector>
using namespace std;
int n;
vector<int> father;
void init(){
    father.resize(n+1);
    for(int i = 0; i <= n; i++){
        father[i] = i;
    }
}
int find(int u){
    if(father[u] == u)return u;
    else father[u] = find(father[u]);
    return father[u];
}
bool isSame(int u, int v){
    u = find(u);
    v = find(v);
    return u == v;
}
void join(int u, int v){
    u = find(u);
    v = find(v);
    if(u == v)return;
    else father[v] = u;
}
int main(){
    int m, source, destination;
    cin >> n >> m;
    init();
    while(m--){
        int s, t;
        cin >> s >> t;
        join(s, t);
    }
    cin >> source >> destination;
    if(isSame(source, destination))cout << 1;
    else cout << 0;
}
```
# 力扣题目
### 寻找图中是否存在路径
```cpp
class Solution {
public:
    vector<int> father;
    void init(const int& n){
        father.resize(n);
        for(int i = 0; i < n; i++){
            father[i] = i;
        }
    }
    int find(int u){
        if(u == father[u])return u;
        else father[u] = find(father[u]);
        return father[u];
    }
    bool isSame(int u, int v){
        u = find(u);
        v = find(v);
        return u == v;
    }
    void join(int u, int v){
        u = find(u);
        v = find(v);
        if(u == v)return;
        father[u] = v;
    }
    bool validPath(int n, vector<vector<int>>& edges, int source, int destination) {
        init(n);
        for(int i = 0; i < edges.size(); i++){
            join(edges[i][0], edges[i][1]);
        }
        if(isSame(source, destination))return true;
        else return false;
    }
};
```

### 冗余连接II
```cpp
class Solution {
public:
vector<int> father;
    void init(const int& n){
        father.resize(n+1);
        for(int i = 0; i <= n; i++){
            father[i] = i;
        }
    }
    int find(int u){
        if(u == father[u])return u;
        else father[u] = find(father[u]);
        return father[u];
    }
    bool isSame(int u, int v){
        u = find(u);
        v = find(v);
        return u == v;
    }
    void join(int u, int v){
        u = find(u);
        v = find(v);
        if(u == v)return;
        father[u] = v;
    }
    bool isTreeAfterMoving(vector<vector<int>>& edges, int deleteEdge){
        init(edges.size());
        for(int i = 0; i < edges.size(); i++){
            if(i == deleteEdge)continue;
            if(isSame(edges[i][0], edges[i][1]))return false;
            join(edges[i][0], edges[i][1]);
        }
        return true;
    }
    vector<int> getRemoveEdge(const vector<vector<int>>& edges){
        init(edges.size());
        for(int i = 0; i < edges.size(); i++){
            if(isSame(edges[i][0], edges[i][1]))return{edges[i][0], edges[i][1]};
            else join(edges[i][0], edges[i][1]);
        }
        return {0, 0};
    }
    vector<int> findRedundantDirectedConnection(vector<vector<int>>& edges) {
        //统计节点入度的数组
        vector<int> inDegree(edges.size()+1, 0);
        for(int i = 0; i < edges.size(); i++){
            inDegree[edges[i][1]]++;
        }
        //记录入度为2的节点，倒叙因为要优先删除最后的边
        vector<int> vec;
        for(int i = edges.size()-1; i >= 0; i--){
            if(inDegree[edges[i][1]] == 2){
                vec.push_back(i);
            }
        }
        if(vec.size() > 0){
            if(isTreeAfterMoving(edges, vec[0]))
                return {edges[vec[0]][0], edges[vec[0]][1]};
            else return {edges[vec[1]][0], edges[vec[1]][1]};
        }
        //有环的情况
        return getRemoveEdge(edges);
    }
};
```
