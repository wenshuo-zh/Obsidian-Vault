# 010|【560】和为k的子串
前缀和+哈希
```cpp
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        int res = 0;
        unordered_map<int, int> mp;//键为前缀和，值为出现次数
        mp[0] = 1;//初始化，当前前缀和本身就等于k时，需要在哈希表找到mp[0] == 1
        //我们要找：当前前缀和 - k = 之前前缀和
        int sum = 0;
        for(int i = 0; i < nums.size(); i++){
            sum += nums[i];//当前前缀和
            if(mp.find(sum - k) != mp.end()){//找到满足的之前前缀和
                res += mp[sum - k];
            }
            mp[sum]++;//存入当前前缀和
        }
        return res;
    }
};
```

# 011|【239】滑动窗口的最大值
单调队列
```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        deque<int>dq;//双向队列，存下标
        vector<int> res;
        for(int i = 0; i < nums.size(); i++){
            //将小于当前值的从队尾删除，
            while(!dq.empty() && nums[dq.back()] < nums[i]){
                dq.pop_back();
            }
            //队首如果超出范围删除
            while(!dq.empty() && dq.front() < i-k+1) dq.pop_front();
            //队尾加入当前值
            dq.push_back(i);
            if(i < k - 1)continue;  
            res.push_back(nums[dq.front()]);  
        }
        return res;
    }
};
```

# 012|【76】最小覆盖子串
```cpp
class Solution {
public:
    string minWindow(string s, string t) {
        unordered_map<char, int>sMp;
        unordered_map<char, int>tMp;
        for(int i = 0; i < t.size(); i++){
            tMp[t[i]]++;
        }
        if(t.empty() || s.size() < t.size()) return "";
        int minCnt = INT_MAX;
        int start = 0;
        int left = 0;
        int valid = 0;
        for(int right = 0; right < s.size(); right++){
            char c = s[right];
            //将right加入窗口
            if(tMp.count(c)){
                sMp[c]++;
                //更新valid
                if(sMp[c] == tMp[c]) valid++;
            }
            while(valid == tMp.size()){
                if(right - left + 1 < minCnt){
                    minCnt = right - left + 1;
                    start = left;
                }
                c = s[left];
                //更新valid
                if(tMp.count(c) && tMp[c] == sMp[c]) valid--;
                //收缩左边界
                if(tMp.count(c)) sMp[c]--;
                left++;
            }
        }
        if(minCnt == INT_MAX) return "";
        return s.substr(start, minCnt);
    }
};
```
