# 链表基础复习
## 节点定义
```cpp
struct ListNode{
	int val;
	ListNode *next;
	ListNode():val(0),next(nullptr){}
};
```
## 尾插法
1. 新节点指向 `nullptr`，因为它将成为新的尾节点。
2. 原尾节点指向新节点。
3. 更新尾指针 `tail`，让它指向新节点。
如果链表为空，开始尾节点为虚头节点
```cpp
// 插入一个值为 1 的节点
ListNode* node = new ListNode(1);
tail->next = node;
tail = node;
```
# 022|【160】相交链表
## 双指针+两链表互换遍历
pA和pB先走自己的链表，走完再走从对方的链表开始走，一定会同时走到同一个节点（因为总路程一样，都是lenA + lenB）
- 如果链表相交了，公共节点相遇
- 否则都是nullptr退出循环
```cpp
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        ListNode *pA = headA, *pB = headB;
        while(pA != pB){
            if(pA != nullptr)pA = pA->next;
            else pA = headB;
            if(pB != nullptr)pB = pB->next;
            else pB = headA;
        }
        return pA;
    }
};
```
# 023|【206】反转链表
## 复制链表头插法
```cpp
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        //res做结果链表的虚头节点
        ListNode* res = new ListNode();
        ListNode* cur = head;
        while(cur){
            //头插法
            ListNode* newCur = new ListNode(cur->val);
            newCur->next = res->next;
            res->next = newCur;
            //遍历原链表
            cur = cur->next;
        }
        return res->next;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(n)
## 三指针原地反转法
```cpp
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode* pre = nullptr;
        ListNode* cur = head;
        ListNode* nxt;
        while(cur){
            nxt = cur->next;//保存后面的节点
            cur->next = pre;//反转
            pre = cur;//前一个节点后移
            cur = nxt;//当前节点后移
        }
        return pre;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)

## 快慢指针找中点+反转后半段链表
```cpp
class Solution {
public:
    bool isPalindrome(ListNode* head) {
        ListNode* slow = new ListNode();
        slow->next = head;
        ListNode* fast = slow;
        while(fast->next && fast->next->next){
            fast = fast->next->next;
            slow = slow->next;
        }
        //反转后半链表
        ListNode* pre = nullptr;
        ListNode* cur = slow->next;
        ListNode* nxt;
        while(cur){
            nxt = cur->next;
            cur->next = pre;
            pre = cur;
            cur = nxt;
        }
        //判断
        ListNode* p = pre;
        while(p){
            if(head->val != p->val)return false;
            head = head->next;
            p = p->next;
        }
        return true;
    }
};
```

# 026|【142】环形链表II
- 快慢指针先相遇，判断链表存在环。
- 一个指针回到头节点，另一个留在相遇点。
- 两个指针每次走一步，再次相遇的位置就是环入口。
```cpp
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        if(!head || !head->next)return NULL;
        ListNode* fast = head;
        ListNode* slow = head;
        bool isCycle = false;
        while(fast->next && fast->next->next){
            fast = fast->next->next;
            slow = slow->next;
            if(fast == slow){
                isCycle = true;
                break;
            }
        }
        if(!isCycle)return NULL;
        ListNode* res = head;
        while(res != slow){
            res = res->next;
            slow = slow->next;
        }
        return res;
    }
};
```

        ListNode* p = list1;
        ListNode* q = list2;

                temp->next = q;
                temp = temp->next;
                q = q->next;
            }
            else{
                temp->next = p;
                temp = temp->next;
                p = p->next;
            }

        else temp->next = q;
        return dummyHead->next;
    }
};

# 028|【2】两数相加
## 模拟竖式加法
```cpp
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode* dummyHead = new ListNode();
        ListNode* tail = dummyHead;
        bool flag = false;
        while(l1 || l2 || flag){
            int val = 0;
            if(l1)val += l1->val;
            if(l2)val += l2->val;
            if(flag) {
                val++;
                flag = false;
            }
            if(val >= 10){
                val -= 10;
                flag = true;
            }
            ListNode* node = new ListNode(val);
            tail->next = node;
            tail = node;
            if(l1)l1 = l1->next;
            if(l2)l2 = l2->next;
        }
        return dummyHead->next;
    }
};
```
# 029|【19】删除链表的倒数第N个节点
## 快慢指针
快指针先走n步，之后两个指针一起走，相差就是n；快指针走到nullptr，慢指针就是删除点
```cpp
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode* dummyHead = new ListNode(0, head);
        ListNode* fast = dummyHead;
        ListNode* slow = dummyHead;
        while(n--)fast = fast->next;
        while(fast->next){
            fast = fast->next;
            slow = slow->next;
        }
        slow->next = slow->next->next;
        return dummyHead->next;
    }
};
```

class Solution {

        if(!head || !head->next)return head;
        //pre是要交换两个节点的先驱节点
        ListNode* pre = dummyHead;
        while(pre->next && pre->next->next){
            ListNode* first = pre->next;
            ListNode* second = first->next;
            ListNode* nextpre = second->next;
            pre->next = second;
            second->next = first;
            first->next = nextpre;
            pre = first;
        }
        return dummyHead->next;
    }

# 031|【25】K个一组翻转链表
```cpp
class Solution {
public:
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode* dummyHead = new ListNode(0, head);
        // pre 始终指向当前分组前面的节点，负责连接前后两组
        ListNode* pre = dummyHead;
        // cur 指向当前分组的第一个节点
        ListNode* cur = pre->next;
        // nxt 暂存 cur 的下一个节点，防止反转时丢失后面的链表
        ListNode* nxt = nullptr;
        while(cur){
            // end 用来检查当前分组是否有 k 个节点
            ListNode* end = pre;
            for(int i = 0; i < k; i++){
                if(end)end = end->next;
            }
            if(!end)break;
            // 保存当前分组原来的前驱节点
            // 反转完成后，需要让它指向反转后的新头节点
            ListNode* start = pre;
            // prev 是反转过程中的前驱节点
            // 初始指向 pre，让当前组第一个节点能够正确连接前面的链表
            ListNode* prev = pre;
            // 保存当前分组原来的第一个节点
            // 反转后它会变成当前分组的最后一个节点
            pre = pre->next;
            // 反转当前分组的 k 个节点
            int n = k;
            while(n--){
                nxt = cur->next;  // 保存后继节点
                cur->next = prev; // 当前节点指向前驱节点
                prev = cur;       // 前驱节点向前移动
                cur = nxt;        // 当前节点向后移动
            }
            // 连接前一部分和反转后的分组
            // prev 此时指向反转后的新头节点
            start->next = prev;
            // 原来的第一个节点现在是分组尾节点
            // 将它连接到下一组的起点
            pre->next = cur;
        }
        return dummyHead->next;
    }
};
```

            Node* newNode = new Node(oriNode->val);
            mp[oriNode] = newNode;
            oriNode = oriNode->next;

    }

# 033|【148】排序链表
## 归并排序
```cpp
class Solution {
public:
    ListNode* split(ListNode* head, int size){
        if(!head)return nullptr;
        ListNode* cur = head;
        for(int i = 1; i < size && cur->next; i++)cur = cur->next;
        ListNode* second = cur->next;
        cur->next = nullptr;
        return second;
    }
    ListNode* merge(ListNode* first, ListNode* second){
        ListNode* merged = new ListNode(0);
        ListNode* cur = merged;
        while(first && second){
            if(first->val > second->val){
                cur->next = second;
                second = second->next;
            }else{
                cur->next = first;
                first = first->next;
            }
            cur = cur->next;
        }
        if(first)cur->next = first;
        else cur->next = second;
        return merged->next;
    }
    ListNode* sortList(ListNode* head) {
        int n = 0;
        ListNode* cur = head;
        while(cur){
            n++;
            cur = cur->next;
        }        
        cur = head;
        ListNode* dummyHead = new ListNode(0, head);
        ListNode* tail = dummyHead;
        for(int i = 1; i < n; i *= 2){
            //每一轮归并都要重置cur和tail
            cur = dummyHead->next;
            tail = dummyHead;
            while(cur){
                ListNode* first = cur;
                ListNode* second = split(cur, i);
                cur = split(second, i);
                tail->next = merge(first, second);
                while(tail->next)tail = tail->next;
            }
        }
        return dummyHead->next;
    }
};
```

# 035|【146】LRU缓存
```cpp
class LRUCache {
public:
    struct Node{
        int key;
        int val;
        Node* next;
        Node* prev;
        Node(int k, int v):key(k), val(v), next(nullptr), prev(nullptr){}
    };
    unordered_map<int, Node*> mp;
    int l_capacity;//容量
    Node* dummyHead;
    Node* dummyTail;
    //删除指定节点
    void removeNode(Node* node){
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }
    //头插新节点
    void addToHead(Node* node){
        node->next = dummyHead->next;
        node->prev = dummyHead;
        dummyHead->next->prev = node;
        dummyHead->next = node;
    }
    //构造函数
    LRUCache(int capacity) {
        this->l_capacity = capacity;
        dummyHead = new Node(0, 0);
        dummyTail = new Node(0, 0);
        dummyHead->next = dummyTail;
        dummyTail->prev = dummyHead; 
    }
    
    int get(int key) {
        if(mp.find(key) == mp.end()) return -1;
        Node* n = mp[key];
        removeNode(n);
        addToHead(n);
        return n->val;
    }
    
    void put(int key, int value) {
        if(mp.find(key) == mp.end()){
            Node* newNode = new Node(key, value);
            mp[key] = newNode;
            addToHead(newNode);
            if(mp.size() > l_capacity){
                Node* last = dummyTail->prev;
                mp.erase(last->key);
                removeNode(last);
                delete last;
            }
        }
        else{
            removeNode(mp[key]);
            mp[key]->val = value;
            addToHead(mp[key]);
        }
    }
};
```
