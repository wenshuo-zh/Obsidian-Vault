# 链表基础复习
## 节点定义
```cpp
struct ListNode{
	int val;
	ListNode *next;
	ListNode():val(0),next(nullptr){}
};
```
## 尾插法
1. 新节点指向 `nullptr`，因为它将成为新的尾节点。
2. 原尾节点指向新节点。
3. 更新尾指针 `tail`，让它指向新节点。
如果链表为空，开始尾节点为虚头节点
```cpp
// 插入一个值为 1 的节点
ListNode* node = new ListNode(1);
tail->next = node;
tail = node;
```
# 021|【160】相交链表
## 双指针+两链表互换遍历
pA和pB先走自己的链表，走完再走从对方的链表开始走，一定会同时走到同一个节点（因为总路程一样，都是lenA + lenB）
- 如果链表相交了，公共节点相遇
- 否则都是nullptr退出循环
```cpp
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        ListNode *pA = headA, *pB = headB;
        while(pA != pB){
            if(pA != nullptr)pA = pA->next;
            else pA = headB;
            if(pB != nullptr)pB = pB->next;
            else pB = headA;
        }
        return pA;
    }
};
```
# 022|【206】反转链表
## 复制链表头插法
```cpp
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        //res做结果链表的虚头节点
        ListNode* res = new ListNode();
        ListNode* cur = head;
        while(cur){
            //头插法
            ListNode* newCur = new ListNode(cur->val);
            newCur->next = res->next;
            res->next = newCur;
            //遍历原链表
            cur = cur->next;
        }
        return res->next;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(n)
## 三指针原地反转法
```cpp
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode* pre = nullptr;
        ListNode* cur = head;
        ListNode* nxt;
        while(cur){
            nxt = cur->next;//保存后面的节点
            cur->next = pre;//反转
            pre = cur;//前一个节点后移
            cur = nxt;//当前节点后移
        }
        return pre;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)
# 023|【234】回文链表
## 快慢指针找中点+反转后半段链表
```cpp
class Solution {
public:
    bool isPalindrome(ListNode* head) {
        ListNode* slow = new ListNode();
        slow->next = head;
        ListNode* fast = slow;
        while(fast->next && fast->next->next){
            fast = fast->next->next;
            slow = slow->next;
        }
        //反转后半链表
        ListNode* pre = nullptr;
        ListNode* cur = slow->next;
        ListNode* nxt;
        while(cur){
            nxt = cur->next;
            cur->next = pre;
            pre = cur;
            cur = nxt;
        }
        //判断
        ListNode* p = pre;
        while(p){
            if(head->val != p->val)return false;
            head = head->next;
            p = p->next;
        }
        return true;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)
# 024|【141】环形链表
## 快慢指针
成环一定会相遇，不成环一定会走到空
```cpp
class Solution {
public:
    bool hasCycle(ListNode *head) {
        if(!head || !head->next)return false;
        ListNode* fast = head;
        ListNode* slow = head;
        while(fast->next && fast->next->next){
            fast = fast->next->next;
            slow = slow->next;
            if(fast == slow){
                return true;
            }
        }
        return false;
    }
};
```
# 025|【142】环形链表II
- 快慢指针先相遇，判断链表存在环。
- 一个指针回到头节点，另一个留在相遇点。
- 两个指针每次走一步，再次相遇的位置就是环入口。
```cpp
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        if(!head || !head->next)return NULL;
        ListNode* fast = head;
        ListNode* slow = head;
        bool isCycle = false;
        while(fast->next && fast->next->next){
            fast = fast->next->next;
            slow = slow->next;
            if(fast == slow){
                isCycle = true;
                break;
            }
        }
        if(!isCycle)return NULL;
        ListNode* res = head;
        while(res != slow){
            res = res->next;
            slow = slow->next;
        }
        return res;
    }
};
```
# 026|【21】合并两个有序链表
```cpp
class Solution {
public:
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        ListNode* dummyHead = new ListNode();
        ListNode* p = list1;
        ListNode* q = list2;
        ListNode* temp = dummyHead;
        while(p && q){
            if(p->val > q->val){
                temp->next = q;
                temp = temp->next;
                q = q->next;
            }
            else{
                temp->next = p;
                temp = temp->next;
                p = p->next;
            }
        }
        if(p)temp->next = p;
        else temp->next = q;
        return dummyHead->next;
    }
};
```
# 027|【2】两数相加
## 模拟竖式加法
```cpp
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode* dummyHead = new ListNode();
        ListNode* tail = dummyHead;
        bool flag = false;
        while(l1 || l2 || flag){
            int val = 0;
            if(l1)val += l1->val;
            if(l2)val += l2->val;
            if(flag) {
                val++;
                flag = false;
            }
            if(val >= 10){
                val -= 10;
                flag = true;
            }
            ListNode* node = new ListNode(val);
            tail->next = node;
            tail = node;
            if(l1)l1 = l1->next;
            if(l2)l2 = l2->next;
        }
        return dummyHead->next;
    }
};
```
# 028|【19】删除链表的倒数第N个节点
## 快慢指针
快指针先走n步，之后两个指针一起走，相差就是n；快指针走到nullptr，慢指针就是删除点
```cpp
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode* dummyHead = new ListNode(0, head);
        ListNode* fast = dummyHead;
        ListNode* slow = dummyHead;
        while(n--)fast = fast->next;
        while(fast->next){
            fast = fast->next;
            slow = slow->next;
        }
        slow->next = slow->next->next;
        return dummyHead->next;
    }
};
```
# 029|【24】两两交换链表中的节点
```cpp
class Solution {
public:
    ListNode* swapPairs(ListNode* head) {
        ListNode* dummyHead = new ListNode(0, head);
        if(!head || !head->next)return head;
        //pre是要交换两个节点的先驱节点
        ListNode* pre = dummyHead;
        while(pre->next && pre->next->next){
            ListNode* first = pre->next;
            ListNode* second = first->next;
            ListNode* nextpre = second->next;
            pre->next = second;
            second->next = first;
            first->next = nextpre;
            pre = first;
        }
        return dummyHead->next;
    }
};
```
