# 013|【53】最大子数组和
## 前缀和解法
```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int sum = 0, minSum = 0, res = INT_MIN;
        for(int i = 0; i < nums.size(); i++){
            sum += nums[i];//当前前缀和
            res = max(res, sum - minSum);//更新最大子数组和
            minSum = min(minSum, sum);//更新最小前缀和
        }
        return res;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)
## 动态规划解法
```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        vector<int> dp(nums.size());//以nums[i]为结尾的最大子数组和
        dp[0] = nums[0];
        int res = nums[0];
        for(int i = 1; i < nums.size(); i++){
            dp[i] = max(nums[i], dp[i-1] + nums[i]);
            res = max(dp[i], res);
        }
        return res;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(n)
## 贪心、dp优化
空间优化后的动态规划
也可以从贪心角度理解：如果之前的连续子数组贡献为负，就不再保留它，重新开始。
```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int res = nums[0];
        int count = 0;
        for(int i = 0; i < nums.size(); i++){
            count = max(nums[i], count + nums[i]);
            res = max(count, res);
        }
        return res;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)

# 015|【189】轮转数组
## 三次反转解法
```cpp
class Solution {
public:
    void rotate(vector<int>& nums, int k) {
        while(k > nums.size()) k %= nums.size();
        int left = 0, right = nums.size()-1;
        while(left < right)swap(nums[left++], nums[right--]);
        left = 0, right = k-1;
        while(left < right)swap(nums[left++], nums[right--]);
        left = k, right = nums.size()-1;
        while(left < right)swap(nums[left++], nums[right--]);                    
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)
# 016|【238】除了自身以外数组的乘积
## 前缀积 \* 后缀积
```cpp
class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        vector<int> frontProduct(nums.size());//前缀积,从nums[0]到nums[i] 的乘积
        vector<int> backProduct(nums.size());//后缀积,从nums[i]到nums[n-1]的乘积
        frontProduct[0] = nums[0];
        for(int i = 1; i < nums.size(); i++)frontProduct[i] = frontProduct[i-1] * nums[i];
        backProduct[nums.size()-1] = nums.back();
        for(int i = nums.size()-2; i >= 0; i--)backProduct[i] = backProduct[i+1] * nums[i];
        vector<int> res(nums.size());
        res[0] = backProduct[1];
        res[nums.size()-1] = frontProduct[nums.size()-2];
        for(int i = 1; i < nums.size()-1; i++){
            res[i] = frontProduct[i-1] * backProduct[i+1];
        }
        return res;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(n)
## 优化空间复杂度版本：
```cpp
class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        int n = nums.size();
        vector<int> res(n);
        res[0] = nums[0];
        //前缀积直接存res数组里
        for(int i = 1; i < n; i++)res[i] = res[i-1] * nums[i];
        //后缀积
        int pro = nums[n-1];
        res[n-1] = res[n-2];
        for(int i = n-2; i > 0; i--){
            res[i] = res[i-1] * pro;
            pro *= nums[i];
        }
        res[0] = pro;
        return res;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)
# 017|【41】缺失的第一个正数
原地哈希+寻找缺失值
```cpp
class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        int curIndex = 0;
        //先判断数字是否在 [1, n] 范围内，合法之后再计算下标，防止下标越界
        while(curIndex < n){
            //值在[1-n]这个范围
            if(nums[curIndex] > 0 && nums[curIndex] <= n){
                int nextIndex = nums[curIndex] - 1;
                //值和下一个要去位置的值不相等
                if(nums[curIndex] != nums[nextIndex]){
                    swap(nums[curIndex], nums[nextIndex]);
                }
                else curIndex++;
            }
            else curIndex++;
        }
        for(int i = 0; i < n; i++){
            if(nums[i] != i + 1)return i + 1;
        }
        return n + 1;
    }
};
```
时间复杂度：O(n)
空间复杂度：O(1)
