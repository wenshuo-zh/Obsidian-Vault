# 001|【1】两数之和
## 暴力解法
```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        vector<int> res(2);
        for(int i = 0; i < nums.size(); i++){
            for(int j = nums.size()-1; j > i; j--){
                if(nums[i] + nums[j] == target){
                    res[0] = i;
                    res[1] = j;
                    return res;
                }
            }
        }
        return res;
    }
};
```
时间复杂度：O(n^2) 
空间复杂度：O(1)
## 哈希表解法
```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        vector<int> res(2);
        unordered_map<int, int> mp;//键存数值，值存下标
        for(int i = 0; i < nums.size(); i++){
            int temp = target - nums[i];
            if(mp.find(temp) != mp.end()){
                res[0] = i;
                res[1] = mp[temp];
                return res;
            }
            mp[nums[i]] = i;
        }
        return res;
    }
};
```
时间复杂度：平均O(n)
空间复杂度：O(n)
# 002|【49】字母异位词分组
## 暴力解法（超时。。。）
仅作为练习记录
```cpp
class Solution {
public:
    bool isSame(string s1, string s2){
        if(s1.size() != s2.size())return false;
        if(s1.size() == 0)return true;
        unordered_map<char, int> mp;//键存字符，值存出现次数
        for(int i = 0; i < s1.size(); i++)mp[s1[i]]++;
        for(int i = 0; i < s2.size(); i++){
            if(mp.find(s2[i]) == mp.end() || mp[s2[i]] == 0)return false;
            mp[s2[i]]--;
        }
        return true;
    }
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        int n = strs.size();
        vector<bool> visited(n, false);
        vector<vector<string>> res;
        for(int i = 0; i < n; i++){
            if(visited[i])continue;
            vector<string> temp;
            temp.push_back(strs[i]);
            visited[i] = true;
            for(int j = n-1; j > i; j--){
                if(visited[j])continue;
                if(isSame(strs[i], strs[j])){
                    temp.push_back(strs[j]);
                    visited[j] = true;
                }
            }
            res.push_back(temp);
        }
        return res;       
    }
};
```
时间复杂度：O(n^2 * k)，n为字符串数量，k为字符串最大长度
空间复杂度：O(n + k)，不计返回结果
## 排序+哈希解法
```cpp
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string, vector<string>> mp;//key存排序后的字符串，value存
        for(auto s : strs){
            string key = s;
            sort(key.begin(), key.end());
            mp[key].push_back(s);
        }
        vector<vector<string>> res;
	        for(auto it : mp){//it是pair<string, vector<string>>
            res.push_back(it.second);
        }
        return res;
    }
};
```
时间复杂度：平均O(n * k log k)，n为字符串数量，k为字符串最大长度
空间复杂度：O(n * k)，包含哈希表中保存的键和值
# 003|【128】最长连续序列
```cpp
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        unordered_set<int> st;
        for(int i = 0; i < nums.size(); i++){
            st.insert(nums[i]);
        }
        int res = 0;
        for(auto num : st){//遍历去重后的st
            if(st.find(num-1) == st.end()){//此时下标i的数字是一个循环的起点
                int count = 1;
                while(st.find(++num) != st.end()){
                    count++;
                }
                res = max(res, count);
            }
        }
        return res;
    }
};
```
时间复杂度：平均O(n)
空间复杂度：O(n)
