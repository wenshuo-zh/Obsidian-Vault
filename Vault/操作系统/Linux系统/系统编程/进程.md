# 创建进程
## fork()函数
`pid_t pid = fork();`
	fork函数创建一个子进程，创建成功父进程拿到返回值>0（子进程pid）、子进程拿到的返回值为0；创建失败返回-1
**获取pid：** 
`getpid()`获取当前进程的pid、`getppid()`获取当前进程的父进程pid
```c
int main(int argc, char *argv[])
{
    pid_t pid = fork();
    if(pid > 0){
        printf("parent,return child pid = %d, pid = %d\n", pid, getpid());
    }
    if(pid == 0){
        printf("child, pid = %d, ppid = %d\n", getpid(), getppid());
    }
    while(1);
    return 0;
}
```

## 写时复制 
父子进程的局部变量、全局变量、堆区空间是不共享的
**局部变量不共享**
![[295cf67131b0f1e213c305e4aa5a5e5d.png|626]]
**堆区空间不共享** 
```c
int main(int argc, char *argv[])
{
    char *str = (char *)malloc(100);
    int pid = fork();
    strcpy(str, "hello");
    if (pid == 0) {
        strcpy(str, "world");
        printf("child str1 = %s\n", str);
    }
    if (pid > 0) {
        sleep(1);
        printf("parent str1 = %s\n", str);
    }
    while(1);
    return 0;
}
```

**打开的文件，共享文件偏移** 
```c
int main(int argc, char *argv[])
{
    int fd = open("./a.txt", O_RDWR);
    int pid = fork();
    if (pid == 0) {
        write(fd, "hello", 5);
        printf("child, fd = %d\n", fd);
    }
    if (pid > 0) {
        sleep(1);
        char buf[1024];
        lseek(fd, 0, SEEK_SET);
        int read_count = read(fd, buf, sizeof(buf));
        buf[read_count] = '\0';
        printf("parent, fd = %d, read_count = %d, buf = %s\n", fd, read_count, buf);
    }
    while (1);
    return 0;
}
```

# 终止进程
## exit()函数
`void exit(int status);`
	status为进程退出状态码
**与return的区别：** 在任意函数中调用exit()，整个进程直接终止，而return仅仅退出当前函数；但在main函数中，`return status;`等价于`exit(status);`
```c
int main(int argc, char *argv[])
{
    printf("exit\n");
    exit(100);
    printf("return\n");
    return 0;
}
```
用`echo $?`查看最近一次进程的返回值

## 孤儿进程
```c
int main(int argc, char *argv[])
{
    pid_t pid = fork();
    if(pid > 0){
        sleep(5);
        printf("parent pid = %d", getpid());
    }
    if(pid == 0){
        while(1){
            printf("child pid = %d, ppid = %d\n", getpid(), getppid());
            sleep(1);
        }
    }
    return 0;
}
```
![[Pasted image 20260926111456.png]]
现象：前五秒：父进程还活着，子进程ppid为父进程pid
       五秒后：父进程退出，子进程变成孤儿进程被`init/systemd`进程收养，子进程ppid为1
       `ctrl+c` 退出不了，新开终端`kill 3295`杀死孤儿进程
## 僵尸进程
子进程结束后，大部分资源已经释放，但内核仍保留PID、退出状态等少量信息；父进程未通过wait/waitpid回收这些信息时，子进程变成僵尸进程
```c
int main(int argc, char *argv[])
{
    pid_t pid = fork();

    if(pid == 0){
        printf("child pid = %d\n", getpid());
        return 1;
    }
    while(1);
    return 0;
}
```

# 回收进程
## wait函数
- 阻塞等待子进程退出
- 回收子进程残留资源
- 获取子进程结束状态
`pid_t wait(int *status)` 
```c
int main(int argc, char *argv[])
{
    pid_t pid = fork();
    if(pid == 0){
        printf("child pid = %d\n", getpid());
        sleep(3);
    }else{
        int wpid = wait(NULL);
        printf("waitpid = %d\n", wpid);
    }
    return 0;
}
```
父进程进行到wait()会阻塞，直到子进程退出
## waitpid函数
`pid_t waitpid(pid_t pid, int *status, int options)`
wait的增强版，可以指定等待的子进程，wait(NULL)等价waitpid(-1, NULL, 0);
核心参数pid 
	`pid > 0`：等待pid=pid的子进程
	`pid = -1`：等待任意子进程
参数options
	`0`：堵塞等待
	`WNOHANG`：非阻塞
### waitpid返回值
- `> 0`：成功回收子进程，返回该子进程的PID
- `0`：指定了`WNOHANG`，且目标子进程还未退出
- `-1`：调用失败，例如没有可等待的子进程
**阻塞等待版本** waitpid主动等 C2，C2 没死，父就一直卡住不动
```c
int main(int argc, char *argv[])
{
    int pid, wid;
    for(int i = 0; i < 5; i++){
        pid = fork();
        if(pid == 0){
            break;//子进程break，为了防止子进程继续走循环导致创出一大堆子进程
        }
        if(i == 2){
            wid = pid;
            printf("i = 2 pid = %d\n", wid);
        }
    }
    if(pid > 0){
        int wpid = waitpid(wid, NULL, 0);
        printf("wpid = %d\n", wpid);
        while(1);
    }else{
        sleep(5);
        return 5;
    }
}
```
![[b85c1cd0db6489f94ca51ec63eefbdf3.png|532]]
![[9f388ed47e2f37cadc51935be2635bec.png]]
	结果：wid开始是未初始化的随机值，父进程`i==2`时，`wid=pid`，然后父进程阻塞等待结束，回收i=2的子进程，而其他子进程变成僵尸进程，因为其他子进程结束了未被父进程回收（图里可以看到四个僵尸进程）
**非阻塞版本**  子进程 1 秒就全部退出，父进程睡 5 秒才醒来； 醒来后用非阻塞 waitpid回收 C2；剩下 4 个子进程保持僵尸
```c
int main(int argc, char *argv[])
{
    int pid, wid;
    for(int i = 0; i < 5; i++){
        pid = fork();
        if(pid == 0){
            break;
        }
        if(i == 2){
            wid = pid;
            printf("i = 2 pid = %d\n", wid);
        }
    }
    if(pid > 0){
        sleep(5);
        int wpid = waitpid(wid, NULL, WNOHANG);
        printf("wpid = %d\n", wpid);
        while(1);
    }else{
        sleep(1);
        return 5;
    }
}
```
